import pandas as pd
from sklearn.model_selection import train_test_split, cross_val_score
from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score

# Toy dataset with numeric + categorical columns
df = pd.DataFrame({
    'age':    [22, 35, 58, 41, 27, 49, 33, 60],
    'salary': [40, 80, 60, 95, 50, 70, 55, 90],
    'dept':   ['eng', 'mkt', 'eng', 'mkt', 'hr',  'eng', 'hr',  'mkt'],
    'bought': [0, 1, 1, 1, 0, 1, 0, 1]
})

X = df.drop('bought', axis=1)
y = df['bought']

numeric = ['age', 'salary']
categorical = ['dept']

# Preprocess: scale numerics, one-hot categoricals
preprocess = ColumnTransformer([
    ('num', StandardScaler(), numeric),
    ('cat', OneHotEncoder(), categorical),
])

# Full pipeline = preprocess + model
pipe = Pipeline([
    ('prep', preprocess),
    ('clf',  RandomForestClassifier(n_estimators=100, random_state=0))
])

# Cross-validation
scores = cross_val_score(pipe, X, y, cv=3)
print('CV accuracy:', scores.mean().round(3))

# Final fit & score
X_train, X_test, y_train, y_test = train_test_split(X, y, random_state=0)
pipe.fit(X_train, y_train)
print('test acc  :', accuracy_score(y_test, pipe.predict(X_test)))

15 Clustering with K-Means

scikit-learn · unsupervised 
import numpy as np
from sklearn.datasets import make_blobs
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score
import matplotlib.pyplot as plt

# Synthetic blobs
X, y_true = make_blobs(n_samples=300, centers=4,
                       cluster_std=0.7, random_state=0)

# Elbow method — pick k
inertias = []
for k in range(2, 8):
    km = KMeans(n_clusters=k, n_init=10, random_state=0)
    km.fit(X)
    inertias.append(km.inertia_)

# Silhouette — quality of clustering
best_k, best_score = 2, -1
for k in range(2, 8):
    labels = KMeans(n_clusters=k, n_init=10,
                    random_state=0).fit_predict(X)
    score = silhouette_score(X, labels)
    if score > best_score:
        best_k, best_score = k, score
print(f'best k = {best_k}  (silhouette = {best_score:.3f})')

# Final model
km = KMeans(n_clusters=best_k, n_init=10, random_state=0)
labels = km.fit_predict(X)
centers = km.cluster_centers_

plt.scatter(X[:, 0], X[:, 1], c=labels, cmap='viridis', s=20)
plt.scatter(centers[:, 0], centers[:, 1], c='red', marker='X', s=200)
plt.title(f'K-Means (k={best_k})')
plt.show()

# A custom transformer that adds ratio features
class RatioAdder(BaseEstimator, TransformerMixin):
    def __init__(self, num_cols=('height', 'weight')):
        self.num_cols = num_cols

    def fit(self, X, y=None):
        return self

    def transform(self, X):
        X = pd.DataFrame(X, columns=self.num_cols)
        out = X.copy()
        out['bmi'] = X['weight'] / (X['height'] / 100) ** 2
        return out.values

            ('ratio', RatioAdder(('height','weight'))),
            ('scale', StandardScaler())
        ]), ['height', 'weight']),
        ('cat', OneHotEncoder(), ['city'])
    ])),
    ('clf', LogisticRegression(max_iter=1000))
])

class TextCleaner(BaseEstimator, TransformerMixin):
    def __init__(self, lower=True, strip_html=True, min_len=2):
        self.lower = lower
        self.strip_html = strip_html
        self.min_len = min_len

    def fit(self, X, y=None):
        return self

    def transform(self, X):
        out = []
        for s in X:
            if not isinstance(s, str):
                s = '' if pd.isna(s) else str(s)
            if self.lower:
                s = s.lower()
            if self.strip_html:
                s = re.sub(r'<[^>]+>', ' ', s)
            s = re.sub(r'http\S+|www\.\S+', ' ', s)      # URLs
            s = re.sub(r'[^a-z0-9\s]+', ' ', s)         # punctuation
            s = re.sub(r'\s+', ' ', s).strip()
            tokens = [t for t in s.split() if len(t) >= self.min_len]
            out.append(' '.join(tokens))
        return out

49 Mini-Project: Churn Pipeline

import numpy as np
import pandas as pd
from sklearn.model_selection import (train_test_split,
                                     StratifiedKFold, cross_val_score)
from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.ensemble import GradientBoostingClassifier
from sklearn.metrics import (roc_auc_score, classification_report,
                             confusion_matrix)

# 1) Synthesize a churn dataset
rng = np.random.default_rng(0)
N = 5000
df = pd.DataFrame({
    'tenure_mo': rng.integers(1, 60, N),
    'monthly':   rng.normal(65, 20, N).round(2),
    'support_calls': rng.integers(0, 8, N),
    'plan':      rng.choice(['basic','plus','pro'], N, p=[.5,.3,.2]),
    'churn':     rng.choice([0,1], N, p=[.78,.22])
})
# Make churn correlated with support_calls & plan
df.loc[df['support_calls'] >= 5, 'churn'] = (
    rng.choice([0,1], size=(df['support_calls'] >= 5).sum(), p=[.3,.7])
)
df.loc[df['plan'] == 'basic', 'churn'] = (
    rng.choice([0,1], size=(df['plan'] == 'basic').sum(), p=[.6,.4])
)

# 2) Split
X = df.drop('churn', axis=1); y = df['churn']
X_tr, X_te, y_tr, y_te = train_test_split(
    X, y, test_size=0.2, random_state=0, stratify=y
)

# 3) Pipeline
num = ['tenure_mo', 'monthly', 'support_calls']
cat = ['plan']
preprocess = ColumnTransformer([
    ('num', StandardScaler(), num),
    ('cat', OneHotEncoder(), cat)
])
pipe = Pipeline([
    ('prep', preprocess),
    ('clf',  GradientBoostingClassifier(
        n_estimators=300, max_depth=3,
        learning_rate=0.05, random_state=0))
])

# 4) CV
cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=0)
cv_auc = cross_val_score(pipe, X_tr, y_tr, cv=cv,
                         scoring='roc_auc', n_jobs=-1)
print(f'CV ROC-AUC: {cv_auc.mean():.3f} +/- {cv_auc.std():.3f}')

# 5) Final fit & test
pipe.fit(X_tr, y_tr)
proba = pipe.predict_proba(X_te)[:, 1]
pred  = (proba >= 0.5).astype(int)
print('test ROC-AUC:', roc_auc_score(y_te, proba).round(3))
print(confusion_matrix(y_te, pred))
print(classification_report(y_te, pred))

# 6) Save the whole pipeline
from joblib import dump
dump(pipe, 'churn_pipeline.joblib')

import pandas as pd
import numpy as np
import time, os

# Build a sizeable frame
N = 1_000_000
df = pd.DataFrame({
    'id':     np.arange(N),
    'value':  np.random.default_rng(0).normal(0, 1, N),
    'bucket': np.random.default_rng(1).choice(list('abcde'), N),
    'flag':   np.random.default_rng(2).integers(0, 2, N)
})
print('frame:', df.shape, df.memory_usage(deep=True).sum() / 1e6, 'MB')

def bench(label, fn):
    t0 = time.perf_counter(); fn(); dt = time.perf_counter() - t0
    print(f'{label:35s} {dt*1000:7.1f} ms')

# 1) CSV round-trip
csv_path = 'big.csv'
bench('csv write', lambda: df.to_csv(csv_path, index=False))
bench('csv read ', lambda: pd.read_csv(csv_path))
print('  csv size :', os.path.getsize(csv_path) / 1e6, 'MB')
os.remove(csv_path)

# 2) Parquet (columnar, compressed) — best for analytics
parq = 'big.parquet'
bench('parquet write', lambda: df.to_parquet(parq, index=False))
bench('parquet read ', lambda: pd.read_parquet(parq))
print('  parq size :', os.path.getsize(parq) / 1e6, 'MB')
os.remove(parq)

# 3) Feather (fast binary, cross-language)
fea = 'big.feather'
bench('feather write', lambda: df.to_feather(fea))
bench('feather read ', lambda: pd.read_feather(fea))
print('  fea size  :', os.path.getsize(fea) / 1e6, 'MB')
os.remove(fea)

# 4) Read in chunks (when the file is too big for RAM)
big = pd.concat([df] * 5, ignore_index=True)   # 5M rows
big.to_csv('huge.csv', index=False)
total = 0
for chunk in pd.read_csv('huge.csv', chunksize=200_000):
    total += (chunk['value'] > 0).sum()
print('chunked sum :', total)
os.remove('huge.csv')

# 5) Pickle — fastest, but Python-only
pk = 'big.pkl'
bench('pickle write', lambda: df.to_pickle(pk))
bench('pickle read ', lambda: pd.read_pickle(pk))
