#39 Integer Right Triangles

For which value of p ≤ 1000, is the number of solutions maximised?

Waiting for input...
// SOURCE CODE
const solve = () => {
    let maxSols = 0;
    let bestP = 0;
    
    for (let p = 1; p <= 1000; p++) {
        let solutions = 0;
        for (let a = 1; a < p / 3; a++) {
            for (let b = a; b < p / 2; b++) {
                const c = p - a - b;
                if (a * a + b * b === c * c) {
                    solutions++;
                }
            }
        }
        if (solutions > maxSols) {
            maxSols = solutions;
            bestP = p;
        }
    }
    return bestP;
};
