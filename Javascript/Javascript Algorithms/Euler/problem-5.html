#5 Smallest Multiple

What is the smallest positive number that is evenly divisible by all of the numbers from 1 to 20?

Waiting for input...
// SOURCE CODE
const solve = () => {
    const gcd = (a, b) => !b ? a : gcd(b, a % b);
    const lcm = (a, b) => (a * b) / gcd(a, b);
    let result = 1;
    for (let i = 1; i <= 20; i++) {
        result = lcm(result, i);
    }
    return result;
};
