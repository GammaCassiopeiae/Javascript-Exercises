async function bucketSort(arr) {
    let buckets = Array.from({ length: 5 }, () => []);
    // 1. Distribute into buckets
    arr.forEach(val => {
        let idx = Math.floor(val / 20);
        buckets[idx].push(val);
    });
    // 2. Sort and concatenate
    return buckets.reduce((acc, bucket) => {
        return acc.concat(bucket.sort((a, b) => a - b));
    }, []);
}
