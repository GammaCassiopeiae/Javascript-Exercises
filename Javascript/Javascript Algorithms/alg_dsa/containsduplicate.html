async function containsDuplicate(arr) {
    const seen = new Set();

    for (let i = 0; i < arr.length; i++) {
        if (seen.has(arr[i])) {
            return true; // Duplicate found
        }
        
        seen.add(arr[i]); // Add unique element to set
    }
    
    return false; // All elements are unique
}
