async function findMin(nums) {
    let left = 0;
    let right = nums.length - 1;

    while (left < right) {
        let mid = Math.floor((left + right) / 2);

        // If mid is greater than right, min must be in right half
        if (nums[mid] > nums[right]) {
            left = mid + 1;
        } 
        // Else, min must be in left half (including mid)
        else {
            right = mid;
        }
    }
    
    return nums[left];
}
