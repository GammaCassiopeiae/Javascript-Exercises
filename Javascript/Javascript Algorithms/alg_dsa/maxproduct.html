async function maxProduct(nums) {
    let maxProd = nums[0];
    let currentMax = nums[0];
    let currentMin = nums[0];

    for (let i = 1; i < nums.length; i++) {
        let temp = currentMax;
        
        // currentMax can be established by current element alone, 
        // multiplying by previous currentMax, or multiplying by previous currentMin (if both negative)
        currentMax = Math.max(nums[i], Math.max(nums[i] * currentMax, nums[i] * currentMin));
        
        // currentMin tracks the lowest (most negative) product which could flip positive later
        currentMin = Math.min(nums[i], Math.min(nums[i] * temp, nums[i] * currentMin));
        
        maxProd = Math.max(maxProd, currentMax);
    }
    
    return maxProd;
}
