async function twoPointersSum(arr, target) {
    let left = 0;
    let right = arr.length - 1;

    while (left < right) {
        let sum = arr[left] + arr[right];
        
        if (sum === target) {
            return [left, right]; // Target pair found
        } else if (sum < target) {
            left++; // Sum too small, move left pointer
        } else {
            right--; // Sum too large, move right pointer
        }
    }
    
    return null; // Target pair not found
}
