01 Bubble Sort

O(n²) · stable · comparison 
// Bubble Sort
void bubbleSort(vector<int>& arr) {
    int n = arr.size();
    for (int i = 0; i < n - 1; ++i) {
        bool swapped = false;
        for (int j = 0; j < n - i - 1; ++j) {
            if (arr[j] > arr[j + 1]) {
                swap(arr[j], arr[j + 1]);
                swapped = true;
            }
        }
        if (!swapped) break;
    }
}

02 Selection Sort

O(n²) · in-place 
// Selection Sort
void selectionSort(vector<int>& arr) {
    int n = arr.size();
    for (int i = 0; i < n - 1; ++i) {
        int minIdx = i;
        for (int j = i + 1; j < n; ++j) {
            if (arr[j] < arr[minIdx]) minIdx = j;
        }
        if (minIdx != i) swap(arr[i], arr[minIdx]);
    }
}

03 Insertion Sort

O(n²) · stable · adaptive 
// Insertion Sort
void insertionSort(vector<int>& arr) {
    int n = arr.size();
    for (int i = 1; i < n; ++i) {
        int key = arr[i];
        int j = i - 1;
        while (j >= 0 && arr[j] > key) {
            arr[j + 1] = arr[j];
            --j;
        }
        arr[j + 1] = key;
    }
}

04 Merge Sort

O(n log n) · stable · divide & conquer 
// Merge Sort
void merge(vector<int>& a, int l, int m, int r) {
    vector<int> L(a.begin() + l, a.begin() + m + 1);
    vector<int> R(a.begin() + m + 1, a.begin() + r + 1);
    int i = 0, j = 0, k = l;
    while (i < (int)L.size() && j < (int)R.size()) {
        a[k++] = (L[i] <= R[j]) ? L[i++] : R[j++];
    }
    while (i < (int)L.size()) a[k++] = L[i++];
    while (j < (int)R.size()) a[k++] = R[j++];
}

void mergeSort(vector<int>& a, int l, int r) {
    if (l >= r) return;
    int m = l + (r - l) / 2;
    mergeSort(a, l, m);
    mergeSort(a, m + 1, r);
    merge(a, l, m, r);
}

05 Quick Sort

O(n log n) avg · in-place 
// Quick Sort (Lomuto partition)
int partition(vector<int>& a, int lo, int hi) {
    int pivot = a[hi];
    int i = lo - 1;
    for (int j = lo; j < hi; ++j) {
        if (a[j] <= pivot) {
            ++i;
            swap(a[i], a[j]);
        }
    }
    swap(a[i + 1], a[hi]);
    return i + 1;
}

void quickSort(vector<int>& a, int lo, int hi) {
    if (lo >= hi) return;
    int p = partition(a, lo, hi);
    quickSort(a, lo, p - 1);
    quickSort(a, p + 1, hi);
}

06 Heap Sort

O(n log n) · in-place 
// Heap Sort
void heapify(vector<int>& a, int n, int i) {
    int largest = i;
    int l = 2 * i + 1, r = 2 * i + 2;
    if (l < n && a[l] > a[largest]) largest = l;
    if (r < n && a[r] > a[largest]) largest = r;
    if (largest != i) {
        swap(a[i], a[largest]);
        heapify(a, n, largest);
    }
}

void heapSort(vector<int>& a) {
    int n = a.size();
    for (int i = n / 2 - 1; i >= 0; --i) heapify(a, n, i);
    for (int i = n - 1; i > 0; --i) {
        swap(a[0], a[i]);
        heapify(a, i, 0);
    }
}

07 Counting Sort

O(n + k) · non-comparison 
// Counting Sort (assumes non-negative ints)
void countingSort(vector<int>& a) {
    if (a.empty()) return;
    int k = *max_element(a.begin(), a.end()) + 1;
    vector<int> count(k, 0);
    for (int x : a) ++count[x];
    int idx = 0;
    for (int v = 0; v < k; ++v) {
        while (count[v]-- > 0) a[idx++] = v;
    }
}

08 Shell Sort

O(n^1.3) typical · gap-based 
// Shell Sort
void shellSort(vector<int>& a) {
    int n = a.size();
    for (int gap = n / 2; gap > 0; gap /= 2) {
        for (int i = gap; i < n; ++i) {
            int temp = a[i];
            int j = i;
            while (j >= gap && a[j - gap] > temp) {
                a[j] = a[j - gap];
                j -= gap;
            }
            a[j] = temp;
        }
    }
}

// Binary Search (iterative) — returns index, or -1
int binarySearch(const vector<int>& a, int target) {
    int lo = 0, hi = (int)a.size() - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;
        if (a[mid] == target) return mid;
        if (a[mid] < target) lo = mid + 1;
        else                  hi = mid - 1;
    }
    return -1;
}

11 Radix Sort

O(n·k) · non-comparison · LSD 
// Radix Sort (LSD) — assumes non-negative ints
void radixSort(vector<int>& a) {
    if (a.empty()) return;
    int maxVal = *max_element(a.begin(), a.end());
    vector<int> buf(a.size());
    for (int exp = 1; maxVal / exp > 0; exp *= 10) {
        vector<int> count(10, 0);
        for (int x : a)           ++count[(x / exp) % 10];
        for (int i = 1; i < 10; ++i) count[i] += count[i - 1];
        for (int i = (int)a.size() - 1; i >= 0; --i) {
            buf[--count[(a[i] / exp) % 10]] = a[i];
        }
        a = buf;
    }
}

12 Bucket Sort

O(n + k) · uniform input 
// Bucket Sort — for values in [0, 1)
void bucketSort(vector<double>& a) {
    int n = a.size();
    if (n <= 0) return;
    vector<vector<double>> buckets(n);
    for (double x : a) {
        int idx = (int)(x * n);
        if (idx == n) --idx;
        buckets[idx].push_back(x);
    }
    int k = 0;
    for (auto& b : buckets) {
        sort(b.begin(), b.end());
        for (double x : b) a[k++] = x;
    }
}

13 Comb Sort

O(n²/2^p) · bubble improvement 
// Comb Sort
void combSort(vector<int>& a) {
    int n = a.size();
    int gap = n;
    bool sorted = false;
    const double shrink = 1.3;
    while (!sorted) {
        gap = (int)(gap / shrink);
        if (gap <= 1) { gap = 1; sorted = true; }
        for (int i = 0; i + gap < n; ++i) {
            if (a[i] > a[i + gap]) {
                swap(a[i], a[i + gap]);
                sorted = false;
            }
        }
    }
}

14 Cocktail Shaker

O(n²) · stable · bidirectional 
// Cocktail Shaker Sort
void cocktailSort(vector<int>& a) {
    int lo = 0, hi = (int)a.size() - 1;
    while (lo < hi) {
        bool swapped = false;
        for (int i = lo; i < hi; ++i) {
            if (a[i] > a[i + 1]) { swap(a[i], a[i + 1]); swapped = true; }
        }
        --hi;
        for (int i = hi; i > lo; --i) {
            if (a[i] < a[i - 1]) { swap(a[i], a[i - 1]); swapped = true; }
        }
        ++lo;
        if (!swapped) break;
    }
}

// Bogo Sort — shuffles until sorted (expectancy: O(n·n!))
#include <random>

    static std::mt19937 gen{ std::random_device{}() };
    return gen;
}
bool isSorted(const vector<int>& a) {
    for (size_t i = 1; i < a.size(); ++i)
        if (a[i] < a[i - 1]) return false;
    return true;

        shuffle(a.begin(), a.end(), rng());
    }
}

16 Ternary Search

O(log₃ n) · unimodal function 
// Ternary Search — finds min of unimodal f on [lo, hi]
double ternarySearch(double lo, double hi,
                     const function<double(double)>& f) {
    for (int i = 0; i < 200; ++i) {
        double m1 = lo + (hi - lo) / 3.0;
        double m2 = hi - (hi - lo) / 3.0;
        if (f(m1) < f(m2)) hi = m2;
        else                lo = m1;
    }
    return (lo + hi) / 2.0;
}

17 Jump Search

O(√n) · sorted input 
// Jump Search — returns index, or -1
int jumpSearch(const vector<int>& a, int target) {
    int n = a.size();
    if (n == 0) return -1;
    int step = (int)sqrt(n);
    int prev = 0;
    while (prev < n && a[min(step, n) - 1] < target) {
        prev = step;
        step += (int)sqrt(n);
        if (prev >= n) return -1;
    }
    for (int i = prev; i < min(step, n); ++i) {
        if (a[i] == target) return i;
    }
    return -1;
}

19 Interpolation Search

O(log log n) · uniform sorted 
// Interpolation Search — returns index, or -1
int interpolationSearch(const vector<int>& a, int target) {
    int lo = 0, hi = (int)a.size() - 1;
    while (lo <= hi && target >= a[lo] && target <= a[hi]) {
        if (lo == hi) return (a[lo] == target) ? lo : -1;
        int pos = lo + ((double)(target - a[lo]) *
                       (hi - lo) / (a[hi] - a[lo]));
        if (a[pos] == target) return pos;
        if (a[pos] <  target) lo = pos + 1;
        else                   hi = pos - 1;
    }
    return -1;
}

20 Fibonacci Search

O(log n) · divide & conquer 
// Fibonacci Search — returns index, or -1
int fibSearch(const vector<int>& a, int target) {
    int n = a.size();
    int fib2 = 0, fib1 = 1, fib = fib1 + fib2;
    while (fib < n) { fib2 = fib1; fib1 = fib; fib = fib1 + fib2; }
    int offset = -1;
    while (fib > 1) {
        int i = min(offset + fib2, n - 1);
        if (a[i] < target)      { fib = fib1;  fib1 = fib2; fib2 = fib - fib1; offset = i; }
        else if (a[i] > target) { fib = fib2;  fib1 -= fib2; fib2 = fib - fib1; }
        else return i;
    }
    return (fib1 && offset + 1 < n && a[offset + 1] == target)
            ? offset + 1 : -1;
}

21 Tim Sort

O(n log n) · stable · hybrid 
// Tim Sort — simplified: insertion-sort runs + merge
static const int RUN = 32;
void insertionRun(vector<int>& a, int l, int r) {
    for (int i = l + 1; i <= r; ++i) {
        int key = a[i], j = i - 1;
        while (j >= l && a[j] > key) { a[j + 1] = a[j]; --j; }
        a[j + 1] = key;
    }
}
void mergeRun(vector<int>& a, int l, int m, int r) {
    vector<int> L(a.begin() + l, a.begin() + m + 1);
    vector<int> R(a.begin() + m + 1, a.begin() + r + 1);
    int i = 0, j = 0, k = l;
    while (i < (int)L.size() && j < (int)R.size())
        a[k++] = (L[i] <= R[j]) ? L[i++] : R[j++];
    while (i < (int)L.size()) a[k++] = L[i++];
    while (j < (int)R.size()) a[k++] = R[j++];
}
void timSort(vector<int>& a) {
    int n = a.size();
    for (int i = 0; i < n; i += RUN)
        insertionRun(a, i, min(i + RUN - 1, n - 1));
    for (int size = RUN; size < n; size *= 2) {
        for (int lo = 0; lo < n; lo += 2 * size) {
            int mid = lo + size - 1;
            int hi  = min(lo + 2 * size - 1, n - 1);
            if (mid < hi) mergeRun(a, lo, mid, hi);
        }
    }
}

22 Intro Sort

O(n log n) · quicksort+heapsort 
// Intro Sort — quicksort with heapsort fallback
static const int DEPTH_LIMIT = 16;
void heapifyIntro(vector<int>& a, int n, int i) {
    int L = 2*i+1, R = 2*i+2, big = i;
    if (L < n && a[L] > a[big]) big = L;
    if (R < n && a[R] > a[big]) big = R;
    if (big != i) { swap(a[i], a[big]); heapifyIntro(a, n, big); }
}
void heapsortIntro(vector<int>& a, int n) {
    for (int i = n/2 - 1; i >= 0; --i) heapifyIntro(a, n, i);
    for (int i = n - 1; i > 0; --i) { swap(a[0], a[i]); heapifyIntro(a, i, 0); }
}
int partIntro(vector<int>& a, int lo, int hi) {
    int p = a[hi], i = lo - 1;
    for (int j = lo; j < hi; ++j)
        if (a[j] <= p) swap(a[++i], a[j]);
    swap(a[i + 1], a[hi]);
    return i + 1;
}
void introSortUtil(vector<int>& a, int lo, int hi, int depth) {
    if (hi - lo < 16) {
        for (int i = lo + 1; i <= hi; ++i) {
            int k = a[i], j = i - 1;
            while (j >= lo && a[j] > k) { a[j+1] = a[j]; --j; }
            a[j+1] = k;
        }
        return;
    }
    if (depth == 0) { heapsortIntro(a + lo, hi - lo + 1); return; }
    int p = partIntro(a, lo, hi);
    introSortUtil(a, lo, p - 1, depth - 1);
    introSortUtil(a, p + 1, hi, depth - 1);
}
void introSort(vector<int>& a) {
    introSortUtil(a, 0, (int)a.size() - 1,
                  2 * (int)log2(max((int)a.size(), 1)));
}

24 Pancake Sort

O(n²) · flip-based 
// Pancake Sort
void flip(vector<int>& a, int k) {
    int i = 0;
    while (i < k) { swap(a[i], a[k]); ++i; --k; }
}
int findMax(const vector<int>& a, int n) {
    int idx = 0;
    for (int i = 1; i < n; ++i) if (a[i] > a[idx]) idx = i;
    return idx;
}
void pancakeSort(vector<int>& a) {
    for (int n = (int)a.size(); n > 1; --n) {
        int mi = findMax(a, n);
        if (mi == n - 1) continue;
        if (mi > 0)      flip(a, mi);
        flip(a, n - 1);
    }
}

25 Cycle Sort

O(n²) · in-place · min writes 
// Cycle Sort — minimizes memory writes
void cycleSort(vector<int>& a) {
    int n = a.size();
    for (int cycleStart = 0; cycleStart < n - 1; ++cycleStart) {
        int item = a[cycleStart];
        int pos  = cycleStart;
        for (int i = cycleStart + 1; i < n; ++i)
            if (a[i] < item) ++pos;
        if (pos == cycleStart) continue;
        while (item == a[pos]) ++pos;
        swap(item, a[pos]);
        while (pos != cycleStart) {
            pos = cycleStart;
            for (int i = cycleStart + 1; i < n; ++i)
                if (a[i] < item) ++pos;
            while (item == a[pos]) ++pos;
            swap(item, a[pos]);
        }
    }
}

26 Strand Sort

O(n²) · stable · sublist merge 
// Strand Sort
void strandSort(vector<int>& a) {
    vector<int> out;
    while (!a.empty()) {
        vector<int> strand;
        auto it = a.begin();
        strand.push_back(*it);
        a.erase(it);
        for (auto jt = a.begin(); jt != a.end(); ) {
            if (*jt >= strand.back()) {
                strand.push_back(*jt);
                jt = a.erase(jt);
            } else ++jt;
        }
        // merge strand into out
        vector<int> merged;
        size_t i = 0, j = 0;
        while (i < out.size() && j < strand.size()) {
            if (out[i] < strand[j]) merged.push_back(out[i++]);
            else                     merged.push_back(strand[j++]);
        }
        while (i < out.size())   merged.push_back(out[i++]);
        while (j < strand.size()) merged.push_back(strand[j++]);
        out = move(merged);
    }
    a = move(out);
}

27 Patience Sort

// Patience Sort (simplified — LIS via patience piles)
int patienceSort(const vector<int>& a) {
    vector<int> piles;
    for (int x : a) {
        auto it = lower_bound(piles.begin(), piles.end(), x);
        if (it == piles.end()) piles.push_back(x);
        else                   *it = x;
    }
    return (int)piles.size(); // length of LIS
}

28 Flash Sort

O(n) avg · distribution 
// Flash Sort
#include <algorithm>
void flashSort(vector<int>& a) {
    int n = a.size();
    if (n <= 1) return;
    int m = max(1, (int)(0.43 * n));
    auto [mn, mx] = minmax_element(a.begin(), a.end());
    if (*mn == *mx) return;
    double c1 = (double)(m - 1) / (*mx - *mn);
    vector<int> L(m, 0);
    for (int x : a) {
        int k = (int)(c1 * (x - *mn));
        ++L[k];
    }
    for (int i = 1; i < m; ++i) L[i] += L[i - 1];
    // permutation pass
    int i = 0, k = m - 1;
    while (i < n) {
        k = (int)(c1 * (a[i] - *mn));
        while (i >= L[k]) { --k; k = max(k, 0); }
        swap(a[i], a[L[k] - 1]);
        --L[k];
    }
    // final insertion sort
    for (int i = 1; i < n; ++i) {
        int key = a[i], j = i - 1;
        while (j >= 0 && a[j] > key) { a[j+1] = a[j]; --j; }
        a[j+1] = key;
    }
}

29 Rotated Array Search

O(log n) · rotated sorted 
// Search in Rotated Sorted Array — returns index, or -1
int rotatedSearch(const vector<int>& a, int target) {
    int lo = 0, hi = (int)a.size() - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;
        if (a[mid] == target) return mid;
        if (a[lo] <= a[mid]) { // left half sorted
            if (target >= a[lo] && target < a[mid]) hi = mid - 1;
            else                                       lo = mid + 1;
        } else {                 // right half sorted
            if (target > a[mid] && target <= a[hi]) lo = mid + 1;
            else                                       hi = mid - 1;
        }
    }
    return -1;
}

30 Meta Binary Search

O(log n) · bitwise probe 
// Meta Binary Search (one-sided, bitwise) — returns index, or -1
int metaBinarySearch(const vector<int>& a, int target) {
    int n = a.size();
    if (n == 0) return -1;
    int lo = 0, hi = n - 1;
    int p  = 1 << (31 - __builtin_clz(n)); // largest power of 2 <= n
    while (p > 0) {
        int mid = lo + p - 1;
        if (mid <= hi && a[mid] == target) return mid;
        if (mid <= hi && a[mid] <  target) lo = mid + 1;
        p >>= 1;
    }
    return -1;
}

31 BFS

// BFS — graph traversal, returns visit order
vector<int> bfs(int n, const vector<vector<int>>& adj, int src) {
    vector<int> order, visited(n, 0);
    queue<int> q;
    q.push(src); visited[src] = 1;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        order.push_back(u);
        for (int v : adj[u]) {
            if (!visited[v]) { visited[v] = 1; q.push(v); }
        }
    }
    return order;
}

void dfs(int u, const vector<vector<int>>& adj,
         vector<int>& visited, vector<int>& order) {

    order.push_back(u);

}

33 Dijkstra

O((V+E) log V) · weighted graph 
// Dijkstra — single-source shortest paths (non-negative weights)
vector<long long> dijkstra(int n, const vector<vector<pair<int,int>>>& adj,
                            int src) {
    const long long INF = LLONG_MAX / 4;
    vector<long long> dist(n, INF);
    dist[src] = 0;
    priority_queue<pair<long long,int>,
                   vector<pair<long long,int>>,
                   greater<pair<long long,int>>> pq;
    pq.push({0, src});
    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (d > dist[u]) continue;
        for (auto [v, w] : adj[u]) {
            if (dist[v] > d + w) {
                dist[v] = d + w;
                pq.push({dist[v], v});
            }
        }
    }
    return dist;
}

34 Bellman-Ford

O(V·E) · handles negative weights 
// Bellman-Ford — single-source shortest paths
// Returns false if a negative cycle is reachable.
bool bellmanFord(int n,
                 const vector<tuple<int,int,int>>& edges,
                 int src, vector<long long>& dist) {
    const long long INF = LLONG_MAX / 4;
    dist.assign(n, INF);
    dist[src] = 0;
    for (int i = 0; i < n - 1; ++i) {
        bool any = false;
        for (auto [u, v, w] : edges) {
            if (dist[u] != INF && dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;
                any = true;
            }
        }
        if (!any) break;
    }
    for (auto [u, v, w] : edges)
        if (dist[u] != INF && dist[v] > dist[u] + w) return false;
    return true;
}

35 Two Sum

O(n) · leetcode #1 · hashmap 
// LeetCode #1 — Two Sum
// Returns indices (i, j) such that nums[i] + nums[j] == target.
vector<int> twoSum(const vector<int>& nums, int target) {
    unordered_map<int, int> seen; // value -> index
    for (int i = 0; i < (int)nums.size(); ++i) {
        int need = target - nums[i];
        auto it = seen.find(need);
        if (it != seen.end()) return {it->second, i};
        seen[nums[i]] = i;
    }
    return {}; // no solution
}

bool isValid(const string& s) {
    unordered_map<char, char> pair_{ {')','('}, {']','['}, {'}','{'} };
    stack<char> st;
    for (char c : s) {
        if (c == '(' || c == '[' || c == '{') st.push(c);

            if (st.empty() || st.top() != pair_[c]) return false;
            st.pop();
        }
    }
    return st.empty();
}

}

// LeetCode #238 — Product of Array Except Self
// answer[i] = product of all nums except nums[i] (no division)
vector<int> productExceptSelf(const vector<int>& nums) {
    int n = nums.size();
    vector<int> ans(n, 1);
    int left = 1;
    for (int i = 0; i < n; ++i) { ans[i] = left; left *= nums[i]; }
    int right = 1;
    for (int i = n - 1; i >= 0; --i) { ans[i] *= right; right *= nums[i]; }
    return ans;
}

int lengthOfLongestSubstring(const string& s) {

    int best = 0, lo = 0;

45 3Sum

O(n²) · leetcode #15 · two pointers 
// LeetCode #15 — 3Sum (unique triplets summing to 0)
vector<vector<int>> threeSum(vector<int>& nums) {
    sort(nums.begin(), nums.end());
    vector<vector<int>> res;
    int n = nums.size();
    for (int i = 0; i < n - 2; ++i) {
        if (i && nums[i] == nums[i - 1]) continue;
        int l = i + 1, r = n - 1;
        while (l < r) {
            int s = nums[i] + nums[l] + nums[r];
            if (s < 0)      ++l;
            else if (s > 0) --r;
            else {
                res.push_back({nums[i], nums[l], nums[r]});
                while (l < r && nums[l] == nums[l + 1]) ++l;
                while (l < r && nums[r] == nums[r - 1]) --r;
                ++l; --r;
            }
        }
    }
    return res;
}

struct TreeNode { int val; TreeNode* left; TreeNode* right; };

vector<int> inorderTraversal(TreeNode* root) {
    vector<int> out;
    stack<TreeNode*> st;
    TreeNode* cur = root;
    while (cur || !st.empty()) {
        while (cur) { st.push(cur); cur = cur->left; }
        cur = st.top(); st.pop();
        out.push_back(cur->val);
        cur = cur->right;
    }
    return out;
}

int rob(const vector<int>& nums) {

    for (int x : nums) {

        prev1 = cur;
    }
    return prev1;
}

// LeetCode #322 — Coin Change (fewest coins, or -1 if impossible)
int coinChange(const vector<int>& coins, int amount) {
    vector<int> dp(amount + 1, INT_MAX);
    dp[0] = 0;
    for (int a = 1; a <= amount; ++a) {
        for (int c : coins) {
            if (c <= a && dp[a - c] != INT_MAX)
                dp[a] = min(dp[a], dp[a - c] + 1);
        }
    }
    return dp[amount] == INT_MAX ? -1 : dp[amount];
}

50 LRU Cache

O(1) · leetcode #146 · list+map 
// LeetCode #146 — LRU Cache (get/put in O(1))
class LRUCache {
    list<pair<int,int>> lst; // {key, value}, front = most recent
    unordered_map<int, list<pair<int,int>>::iterator> idx;
    int cap;
public:
    LRUCache(int capacity) : cap(capacity) {}
    int get(int key) {
        if (!idx.count(key)) return -1;
        lst.splice(lst.begin(), lst, idx[key]);
        return idx[key]->second;
    }
    void put(int key, int value) {
        if (idx.count(key)) {
            idx[key]->second = value;
            lst.splice(lst.begin(), lst, idx[key]);
            return;
        }
        if ((int)lst.size() == cap) {
            idx.erase(lst.back().first);
            lst.pop_back();
        }
        lst.push_front({key, value});
        idx[key] = lst.begin();
    }
};
