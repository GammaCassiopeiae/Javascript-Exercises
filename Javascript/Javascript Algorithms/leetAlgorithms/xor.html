⊕ XOR Operations

// Find Single Number (all others appear twice)
function singleNumber(nums) {
    let result = 0;
    for (let num of nums) {
        result ^= num;
    }
    return result;
}

// Find Two Single Numbers
function singleNumber2(nums) {
    let xor = 0;
    for (let num of nums) xor ^= num;
    
    const bit = xor & (-xor);
    
    let num1 = 0, num2 = 0;
    for (let num of nums) {
        if (num & bit) {
            num1 ^= num;
        } else {
            num2 ^= num;
        }
    }
    
    return [num1, num2];
}

// XOR Properties:
// a ^ a = 0
// a ^ 0 = a
// a ^ b = b ^ a (commutative)
// (a ^ b) ^ c = a ^ (b ^ c) (associative)

// Time: O(n), Space: O(1)
