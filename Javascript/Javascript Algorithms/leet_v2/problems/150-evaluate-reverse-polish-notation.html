Evaluate Reverse Polish Notation

var evalRPN = function(tokens) {
    const stack = [];
    const ops = {
        '+': (a, b) => a + b,
        '-': (a, b) => a - b,
        '*': (a, b) => a * b,
        '/': (a, b) => Math.trunc(a / b)
    };
    for (let t of tokens) {
        if (ops[t]) {
            const b = stack.pop(), a = stack.pop();
            stack.push(ops[t](a, b));
        } else stack.push(Number(t));
    }
    return stack[0];
};
