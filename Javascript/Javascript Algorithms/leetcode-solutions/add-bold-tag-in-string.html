Add Bold Tag in String

LeetCode Problem - String Manipulation & Interval Merging

Problem Description

Given a string s and an array of strings words, add bold tags around substrings that exist in words. Overlapping or consecutive substrings should be merged into a single bold section.

JavaScript Solution

function addBoldTag(s, words) {
    const n = s.length;
    const bold = new Array(n).fill(false);
    
    // Mark characters that should be bold
    for (let i = 0; i < n; i++) {
        for (const word of words) {
            if (s.startsWith(word, i)) {
                for (let j = i; j < i + word.length; j++) {
                    bold[j] = true;
                }
            }
        }
    }
    
    // Build result with bold tags
    const result = [];
    let inBold = false;
    
    for (let i = 0; i < n; i++) {
        if (bold[i] && !inBold) {
            result.push('<b>');
            inBold = true;
        } else if (!bold[i] && inBold) {
            result.push('</b>');
            inBold = false;
        }
        result.push(s[i]);
    }
    
    if (inBold) {
        result.push('</b>');
    }
    
    return result.join('');
}

// Usage Example:
const s = "abcxyz123";
const words = ["abc", "xyz"];
console.log(addBoldTag(s, words)); // "<b>abc</b><b>xyz</b>123"
