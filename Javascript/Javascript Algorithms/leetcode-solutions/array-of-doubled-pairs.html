Array of Doubled Pairs

LeetCode Problem - Hash Map + Greedy

Problem Description

Given an integer array arr, return true if you can reorder it such that arr[2*i+1] = 2 * arr[2*i] for every 0 <= i < len(arr)/2.

JavaScript Solution

function canReorderDoubled(arr) {
    const count = new Map();
    
    // Count frequencies
    for (const num of arr) {
        count.set(num, (count.get(num) || 0) + 1);
    }
    
    // Sort by absolute value
    arr.sort((a, b) => Math.abs(a) - Math.abs(b));
    
    for (const num of arr) {
        if (count.get(num) === 0) continue;
        
        const target = num * 2;
        if ((count.get(target) || 0) === 0) {
            return false;
        }
        
        count.set(num, count.get(num) - 1);
        count.set(target, count.get(target) - 1);
    }
    
    return true;
}

// Usage:
console.log(canReorderDoubled([3,1,3,6])); // false
console.log(canReorderDoubled([4,-2,2,-4])); // true
