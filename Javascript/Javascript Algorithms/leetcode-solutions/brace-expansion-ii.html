Brace Expansion II

LeetCode Problem - Stack + Set Operations

Problem Description

Given an expression representing a set of words under brace expansion, return all possible words in sorted order.

JavaScript Solution

function braceExpansionII(expression) {
    const stack = [];
    let seen = new Set();
    let current = new Set();
    let i = 0;
    
    function parse() {
        let result = new Set();
        
        while (i < expression.length) {
            const c = expression[i];
            
            if (c === '{') {
                i++;
                const inner = parse();
                const newResult = new Set();
                for (const a of result.size ? result : ['']) {
                    for (const b of inner) {
                        newResult.add(a + b);
                    }
                }
                result = newResult;
            } else if (c === '}') {
                i++;
                return result;
            } else if (c === ',') {
                i++;
                for (const item of result) seen.add(item);
                result = new Set();
            } else {
                let word = '';
                while (i < expression.length && /[a-z]/.test(expression[i])) {
                    word += expression[i++];
                }
                const newResult = new Set();
                for (const a of result.size ? result : ['']) {
                    newResult.add(a + word);
                }
                result = newResult;
            }
        }
        
        for (const item of result) seen.add(item);
        return seen;
    }
    
    parse();
    return [...seen].sort();
}

// Usage:
console.log(braceExpansionII("{a,b}{c,{d,e}}")); // ["ac","ad","ae","bc","bd","be"]
