Check If It Is a Straight Line

LeetCode Problem - Geometry & Slope Calculation

Problem Description

Given an array of coordinates where coordinates[i] = [x, y], check if all points form a straight line in the XY plane.

JavaScript Solution

function checkStraightLine(coordinates) {
    if (coordinates.length <= 2) return true;
    
    const [x0, y0] = coordinates[0];
    const [x1, y1] = coordinates[1];
    
    // Calculate delta (avoid division for precision)
    const dx = x1 - x0;
    const dy = y1 - y0;
    
    // Check if all other points have same slope
    for (let i = 2; i < coordinates.length; i++) {
        const [x, y] = coordinates[i];
        
        // Cross multiplication to avoid division
        // (y - y0) / (x - x0) === dy / dx
        if (dy * (x - x0) !== dx * (y - y0)) {
            return false;
        }
    }
    
    return true;
}

// Usage Example:
const coords = [[1,2],[2,3],[3,4],[4,5]];
console.log(checkStraightLine(coords)); // true
