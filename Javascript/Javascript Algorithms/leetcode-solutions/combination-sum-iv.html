Combination Sum IV

LeetCode Problem - Dynamic Programming

Problem Description

Given an array of distinct integers nums and a target integer target, return the number of possible combinations that add up to target. Different sequences are counted as different combinations.

JavaScript Solution

function combinationSum4(nums, target) {
    const dp = new Array(target + 1).fill(0);
    dp[0] = 1; // One way to make 0
    
    for (let i = 1; i <= target; i++) {
        for (const num of nums) {
            if (i >= num) {
                dp[i] += dp[i - num];
            }
        }
    }
    
    return dp[target];
}

// Usage Example:
const nums = [1, 2, 3];
const target = 4;
console.log(combinationSum4(nums, target)); // 7
// Possible: (1,1,1,1), (1,1,2), (1,2,1), (1,3), (2,1,1), (2,2), (3,1)
