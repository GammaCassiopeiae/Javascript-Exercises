Compare Version Numbers

LeetCode Problem - String Parsing

Problem Description

Given two version numbers version1 and version2, compare them. Return -1 if version1 < version2, 1 if version1 > version2, 0 if equal.

JavaScript Solution

function compareVersion(version1, version2) {
    const v1 = version1.split('.').map(Number);
    const v2 = version2.split('.').map(Number);
    
    const maxLen = Math.max(v1.length, v2.length);
    
    for (let i = 0; i < maxLen; i++) {
        // Treat missing revisions as 0
        const num1 = v1[i] || 0;
        const num2 = v2[i] || 0;
        
        if (num1 < num2) return -1;
        if (num1 > num2) return 1;
    }
    
    return 0;
}

// Usage Example:
console.log(compareVersion("1.01", "1.001")); // 0 (equal)
console.log(compareVersion("1.0", "1.0.0")); // 0 (equal)
console.log(compareVersion("0.1", "1.1")); // -1
console.log(compareVersion("1.2", "1.10")); // -1
