Convert Sorted Array to BST

LeetCode Problem - Divide and Conquer Tree Building

Problem Description

Given an integer array nums sorted in ascending order, convert it to a height-balanced binary search tree.

JavaScript Solution

class TreeNode {
    constructor(val, left = null, right = null) {
        this.val = val;
        this.left = left;
        this.right = right;
    }
}

function sortedArrayToBST(nums) {
    function build(left, right) {
        if (left > right) return null;
        
        const mid = Math.floor((left + right) / 2);
        const node = new TreeNode(nums[mid]);
        node.left = build(left, mid - 1);
        node.right = build(mid + 1, right);
        return node;
    }
    
    return build(0, nums.length - 1);
}

// Helper: In-order traversal to verify
function inorder(root) {
    if (!root) return [];
    return [...inorder(root.left), root.val, ...inorder(root.right)];
}

// Usage:
const root = sortedArrayToBST([-10,-3,0,5,9]);
console.log(inorder(root)); // [-10, -3, 0, 5, 9]
