Count Largest Group

LeetCode Problem - Digit Sum + Hash Map

Problem Description

Given an integer n, group numbers from 1 to n by sum of their digits. Return the count of groups that have the largest size.

JavaScript Solution

function countLargestGroup(n) {
    const groups = new Map();
    
    function digitSum(num) {
        let sum = 0;
        while (num > 0) {
            sum += num % 10;
            num = Math.floor(num / 10);
        }
        return sum;
    }
    
    // Group by digit sum
    for (let i = 1; i <= n; i++) {
        const sum = digitSum(i);
        groups.set(sum, (groups.get(sum) || 0) + 1);
    }
    
    // Find max size
    let maxSize = 0;
    for (const size of groups.values()) {
        maxSize = Math.max(maxSize, size);
    }
    
    // Count groups with max size
    let count = 0;
    for (const size of groups.values()) {
        if (size === maxSize) count++;
    }
    
    return count;
}

// Usage:
console.log(countLargestGroup(13)); // 4 (groups: [1,10], [2,11], [3,12], [4,13], [5], [6], [7], [8], [9])
