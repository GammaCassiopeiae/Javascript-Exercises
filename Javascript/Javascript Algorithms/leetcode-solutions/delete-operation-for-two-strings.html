Delete Operation for Two Strings

LeetCode Problem - Longest Common Subsequence

Problem Description

Given two strings word1 and word2, return the minimum number of steps required to make them the same by deleting characters.

JavaScript Solution

function minDistance(word1, word2) {
    const m = word1.length, n = word2.length;
    
    // Find LCS length
    const dp = new Array(m + 1).fill(0)
        .map(() => new Array(n + 1).fill(0));
    
    for (let i = 1; i <= m; i++) {
        for (let j = 1; j <= n; j++) {
            if (word1[i - 1] === word2[j - 1]) {
                dp[i][j] = dp[i - 1][j - 1] + 1;
            } else {
                dp[i][j] = Math.max(dp[i - 1][j], dp[i][j - 1]);
            }
        }
    }
    
    const lcs = dp[m][n];
    
    // Delete characters not in LCS from both strings
    return (m - lcs) + (n - lcs);
}

// Usage:
console.log(minDistance("sea", "eat")); // 2 (delete 's' and 't')
