Design Excel Sum Formula

LeetCode Problem - Hash Map + Dependency Tracking

Problem Description

Design a simplified Excel with basic operations: set, get, and sum formula that auto-updates when dependencies change.

JavaScript Solution

class Excel {
    constructor(height, width) {
        this.grid = new Map();
        this.formulas = new Map();
        this.rows = height;
        this.cols = width;
    }

    set(row, col, val) {
        const key = `${row}${col}`;
        this.formulas.delete(key);
        this.grid.set(key, val);
    }

    get(row, col) {
        const key = `${row}${col}`;
        if (this.formulas.has(key)) {
            return this.evaluate(this.formulas.get(key));
        }
        return this.grid.get(key) || 0;
    }

    sum(row, col, numbers) {
        const key = `${row}${col}`;
        this.formulas.set(key, numbers);
        const val = this.evaluate(numbers);
        this.grid.set(key, val);
        return val;
    }

    evaluate(formulas) {
        let sum = 0;
        for (const formula of formulas) {
            if (formula.includes(':')) {
                const [start, end] = formula.split(':');
                const [c1, r1] = this.parseCell(start);
                const [c2, r2] = this.parseCell(end);
                for (let r = r1; r <= r2; r++) {
                    for (let c = c1; c <= c2; c++) {
                        sum += this.get(r, c);
                    }
                }
            } else {
                const [c, r] = this.parseCell(formula);
                sum += this.get(r, c);
            }
        }
        return sum;
    }

    parseCell(cell) {
        const col = cell.charCodeAt(0) - 'A'.charCodeAt(0);
        const row = parseInt(cell.slice(1));
        return [col, row];
    }
}

// Usage:
const excel = new Excel(3, "C");
excel.set(1, "A", 2);
excel.sum(3, "A", ["A1", "A2"]);
