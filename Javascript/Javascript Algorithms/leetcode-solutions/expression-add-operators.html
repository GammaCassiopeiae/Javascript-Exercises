Expression Add Operators

LeetCode Problem - Backtracking with Expression Evaluation

Problem Description

Given a string num containing digits and a target, return all possibilities to add binary operators (+, -, *) so that the expression evaluates to target.

JavaScript Solution

function addOperators(num, target) {
    const result = [];
    
    function backtrack(idx, path, value, last) {
        if (idx === num.length) {
            if (value === target) result.push(path);
            return;
        }
        
        for (let i = idx; i < num.length; i++) {
            // Skip numbers with leading zero
            if (i > idx && num[idx] === '0') break;
            
            const curr = parseInt(num.slice(idx, i + 1));
            
            if (idx === 0) {
                backtrack(i + 1, path + curr, curr, curr);
            } else {
                backtrack(i + 1, path + '+' + curr, value + curr, curr);
                backtrack(i + 1, path + '-' + curr, value - curr, -curr);
                backtrack(i + 1, path + '*' + curr, value - last + last * curr, last * curr);
            }
        }
    }
    
    backtrack(0, '', 0, 0);
    return result;
}

// Usage:
console.log(addOperators("123", 6)); // ["1+2+3","1*2*3"]
