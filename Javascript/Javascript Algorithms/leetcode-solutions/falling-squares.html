Falling Squares

LeetCode Problem - Coordinate Compression + Simulation

Problem Description

Given positions where positions[i] = [left, sideLength], simulate squares falling and return the maximum height after each square drops.

JavaScript Solution

function fallingSquares(positions) {
    const heights = []; // [left, right, height]
    const result = [];
    let maxH = 0;
    
    for (const [left, side] of positions) {
        const right = left + side;
        let baseH = 0;
        
        // Find max height in range
        for (const [l, r, h] of heights) {
            if (l < right && r > left) {
                baseH = Math.max(baseH, h);
            }
        }
        
        const newH = baseH + side;
        heights.push([left, right, newH]);
        maxH = Math.max(maxH, newH);
        result.push(maxH);
    }
    
    return result;
}

// Usage:
console.log(fallingSquares([[1,2],[2,3],[6,1]])); // [2,5,5]
