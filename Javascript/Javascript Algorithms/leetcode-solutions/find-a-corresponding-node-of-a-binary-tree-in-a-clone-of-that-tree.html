Find Corresponding Node in Clone

LeetCode Problem - Tree Traversal

Problem Description

Given two binary trees original and cloned and a reference to a node target in the original tree, return a reference to the same node in the cloned tree.

JavaScript Solution

class TreeNode {
    constructor(val, left = null, right = null) {
        this.val = val;
        this.left = left;
        this.right = right;
    }
}

function getTargetCopy(original, cloned, target) {
    if (!original) return null;
    
    // Found target in original
    if (original === target) return cloned;
    
    // Search in left subtree
    const leftResult = getTargetCopy(original.left, cloned.left, target);
    if (leftResult) return leftResult;
    
    // Search in right subtree
    return getTargetCopy(original.right, cloned.right, target);
}

// Iterative approach (no recursion)
function getTargetCopyIterative(original, cloned, target) {
    const stack = [[original, cloned]];
    
    while (stack.length > 0) {
        const [orig, copy] = stack.pop();
        
        if (orig === target) return copy;
        
        if (orig.left) stack.push([orig.left, copy.left]);
        if (orig.right) stack.push([orig.right, copy.right]);
    }
    
    return null;
}

// Usage:
const original = new TreeNode(7, new TreeNode(4), new TreeNode(3));
const cloned = new TreeNode(7, new TreeNode(4), new TreeNode(3));
console.log(getTargetCopy(original, cloned, original.left)); // cloned.left
