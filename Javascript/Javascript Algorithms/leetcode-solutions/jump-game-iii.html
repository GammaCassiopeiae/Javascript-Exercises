Jump Game III

LeetCode Problem - BFS/DFS Graph Traversal

Problem Description

Given an array of non-negative integers arr, you start at index start. From index i, you can jump to i + arr[i] or i - arr[i]. Check if you can reach any index with value 0.

JavaScript Solution

function canReach(arr, start) {
    const visited = new Set();
    const queue = [start];
    
    while (queue.length > 0) {
        const idx = queue.shift();
        
        if (arr[idx] === 0) return true;
        if (visited.has(idx)) continue;
        
        visited.add(idx);
        
        const left = idx - arr[idx];
        const right = idx + arr[idx];
        
        if (left >= 0) queue.push(left);
        if (right < arr.length) queue.push(right);
    }
    
    return false;
}

// Usage:
console.log(canReach([4,2,3,0,3,1,2], 5)); // true
