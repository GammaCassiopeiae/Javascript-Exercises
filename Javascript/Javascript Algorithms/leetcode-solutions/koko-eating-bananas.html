Koko Eating Bananas

LeetCode Problem - Binary Search

Problem Description

Koko loves bananas. There are n piles of bananas. Koko has h hours to eat all bananas. Find minimum k (bananas per hour) such that Koko can eat all bananas within h hours.

JavaScript Solution

function minEatingSpeed(piles, h) {
    let left = 1;
    let right = Math.max(...piles);
    
    function canFinish(k) {
        let hours = 0;
        for (const pile of piles) {
            hours += Math.ceil(pile / k);
        }
        return hours <= h;
    }
    
    while (left < right) {
        const mid = Math.floor((left + right) / 2);
        
        if (canFinish(mid)) {
            right = mid;
        } else {
            left = mid + 1;
        }
    }
    
    return left;
}

// Usage Example:
const piles = [3, 6, 7, 11];
const h = 8;
console.log(minEatingSpeed(piles, h)); // 4
