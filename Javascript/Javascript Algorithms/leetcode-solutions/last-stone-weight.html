Last Stone Weight

LeetCode Problem - Max Heap Simulation

Problem Description

Given stones where each turn we choose two heaviest and smash them. Return the weight of the last remaining stone.

JavaScript Solution

function lastStoneWeight(stones) {
    // Sort in descending order
    stones.sort((a, b) => b - a);
    
    while (stones.length > 1) {
        const first = stones.shift();
        const second = stones.shift();
        
        if (first !== second) {
            // Insert remaining stone in sorted position
            const diff = first - second;
            let i = 0;
            while (i < stones.length && stones[i] >= diff) i++;
            stones.splice(i, 0, diff);
        }
    }
    
    return stones[0] || 0;
}

// Usage:
console.log(lastStoneWeight([2,7,4,1,8,1])); // 1
