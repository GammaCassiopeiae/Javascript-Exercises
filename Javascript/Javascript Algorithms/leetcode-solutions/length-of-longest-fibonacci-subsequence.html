Length of Longest Fibonacci Subsequence

LeetCode Problem - Dynamic Programming + Hash Set

Problem Description

Given a strictly increasing array of positive integers, find the length of the longest Fibonacci-like subsequence.

JavaScript Solution

function lenLongestFibSubseq(arr) {
    const set = new Set(arr);
    let maxLen = 0;
    
    for (let i = 0; i < arr.length; i++) {
        for (let j = i + 1; j < arr.length; j++) {
            let a = arr[i], b = arr[j];
            let len = 2;
            
            while (set.has(a + b)) {
                const temp = a + b;
                a = b;
                b = temp;
                len++;
            }
            
            if (len > 2) {
                maxLen = Math.max(maxLen, len);
            }
        }
    }
    
    return maxLen;
}

// Usage:
console.log(lenLongestFibSubseq([1,2,3,4,5,6,7,8])); // 5 ([1,2,3,5,8])
