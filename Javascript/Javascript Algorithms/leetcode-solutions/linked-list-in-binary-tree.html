Linked List In Binary Tree

LeetCode Problem - DFS Path Matching

Problem Description

Given a binary tree root and a linked list, return true if all elements in the linked list starting from the head correspond to some downward path connected in the binary tree, otherwise return false.

JavaScript Solution

// TreeNode Definition
class TreeNode {
    constructor(val, left = null, right = null) {
        this.val = val;
        this.left = left;
        this.right = right;
    }
}

// ListNode Definition
class ListNode {
    constructor(val, next = null) {
        this.val = val;
        this.next = next;
    }
}

function isSubPath(head, root) {
    if (root === null) return false;
    
    // Check if path starts at current node, or try left/right subtrees
    return dfs(head, root) || 
           isSubPath(head, root.left) || 
           isSubPath(head, root.right);
}

function dfs(head, node) {
    // Reached end of linked list - success
    if (head === null) return true;
    
    // Reached end of tree path or values don't match - failure
    if (node === null || head.val !== node.val) return false;
    
    // Continue DFS on left or right child
    return dfs(head.next, node.left) || dfs(head.next, node.right);
}

// Helper: Build linked list from array
function buildLinkedList(arr) {
    if (arr.length === 0) return null;
    const head = new ListNode(arr[0]);
    let current = head;
    for (let i = 1; i < arr.length; i++) {
        current.next = new ListNode(arr[i]);
        current = current.next;
    }
    return head;
}

// Usage Example:
const head = buildLinkedList([4, 2, 8]);
const root = new TreeNode(1, 
    new TreeNode(4, new TreeNode(2, new TreeNode(7))),
    new TreeNode(4, new TreeNode(2, new TreeNode(1), new TreeNode(8)))
);
console.log(isSubPath(head, root)); // true or false
