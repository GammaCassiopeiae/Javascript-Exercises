Longest Arithmetic Sequence

LeetCode Problem - Dynamic Programming

Problem Description

Given an array of integers, return the length of the longest arithmetic subsequence. An arithmetic sequence has the same difference between consecutive elements.

JavaScript Solution

function longestArithSeqLength(nums) {
    const n = nums.length;
    const dp = new Array(n).fill(null)
        .map(() => new Map());
    
    let maxLen = 1;
    
    for (let i = 0; i < n; i++) {
        for (let j = 0; j < i; j++) {
            const diff = nums[i] - nums[j];
            
            // Get length from j or default to 1
            const len = (dp[j].get(diff) || 1) + 1;
            
            dp[i].set(diff, len);
            maxLen = Math.max(maxLen, len);
        }
    }
    
    return maxLen;
}

// Usage Example:
const nums = [3, 6, 9, 12];
console.log(longestArithSeqLength(nums)); // 4 (entire array)
