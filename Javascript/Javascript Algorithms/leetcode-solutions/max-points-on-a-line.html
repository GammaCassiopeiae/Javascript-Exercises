Max Points on a Line

LeetCode Problem - Geometry + Hash Map

Problem Description

Given an array of points where points[i] = [xi, yi], return the maximum number of points that lie on the same straight line.

JavaScript Solution

function maxPoints(points) {
    if (points.length <= 2) return points.length;
    
    let max = 0;
    
    for (let i = 0; i < points.length; i++) {
        const slopes = new Map();
        let duplicate = 1;
        let localMax = 0;
        
        for (let j = i + 1; j < points.length; j++) {
            const dx = points[j][0] - points[i][0];
            const dy = points[j][1] - points[i][1];
            
            if (dx === 0 && dy === 0) {
                duplicate++;
                continue;
            }
            
            // Normalize slope using GCD
            const gcdVal = gcd(dx, dy);
            const slope = `${dx/gcdVal}/${dy/gcdVal}`;
            
            slopes.set(slope, (slopes.get(slope) || 0) + 1);
            localMax = Math.max(localMax, slopes.get(slope));
        }
        
        max = Math.max(max, localMax + duplicate);
    }
    
    return max;
}

function gcd(a, b) {
    return b === 0 ? a : gcd(b, a % b);
}

// Usage:
console.log(maxPoints([[1,1],[2,2],[3,3]])); // 3
