Maximum Number of Events That Can Be Attended

LeetCode Problem - Greedy + Min-Heap

Problem Description

Given an array of events where events[i] = [startDay, endDay], find the maximum number of events you can attend. You can only attend one event per day.

JavaScript Solution

function maxEvents(events) {
    // Sort by start day
    events.sort((a, b) => a[0] - b[0]);
    
    const minHeap = [];
    let day = 0, idx = 0, count = 0;
    
    while (idx < events.length || minHeap.length > 0) {
        // Add all events starting today
        if (minHeap.length === 0) {
            day = events[idx][0];
        }
        
        while (idx < events.length && events[idx][0] <= day) {
            minHeap.push(events[idx][1]);
            minHeap.sort((a, b) => a - b);
            idx++;
        }
        
        // Remove expired events
        while (minHeap.length > 0 && minHeap[0] < day) {
            minHeap.shift();
        }
        
        // Attend event with earliest end day
        if (minHeap.length > 0) {
            minHeap.shift();
            count++;
        }
        
        day++;
    }
    
    return count;
}

// Usage:
console.log(maxEvents([[1,2],[2,3],[3,4]])); // 3
