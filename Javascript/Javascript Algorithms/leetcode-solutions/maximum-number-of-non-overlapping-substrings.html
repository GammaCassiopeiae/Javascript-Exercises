Maximum Number of Non-Overlapping Substrings

LeetCode Problem - Greedy + Interval Processing

Problem Description

Given a string of lowercase letters, find maximum number of non-overlapping substrings such that each substring contains all occurrences of every character it includes.

JavaScript Solution

function maxNumOfSubstrings(s) {
    const first = new Map(), last = new Map();
    
    // Find first and last occurrence of each char
    for (let i = 0; i < s.length; i++) {
        if (!first.has(s[i])) first.set(s[i], i);
        last.set(s[i], i);
    }
    
    const intervals = [];
    
    // For each character, find valid substring starting at its first occurrence
    for (const [char, start] of first) {
        let end = last.get(char);
        let valid = true;
        
        for (let i = start; i <= end && valid; i++) {
            if (first.get(s[i]) < start) {
                valid = false;
            }
            end = Math.max(end, last.get(s[i]));
        }
        
        if (valid) intervals.push([start, end]);
    }
    
    // Sort by end position and greedily select
    intervals.sort((a, b) => a[1] - b[1]);
    
    const result = [];
    let prevEnd = -1;
    
    for (const [start, end] of intervals) {
        if (start > prevEnd) {
            result.push(s.slice(start, end + 1));
            prevEnd = end;
        }
    }
    
    return result;
}

// Usage:
console.log(maxNumOfSubstrings("adefaddaccc")); // ["e","f","ccc"]
