Maximum Points from Cards

LeetCode Problem - Sliding Window

Problem Description

Given an array of card points and integer k. You can take exactly k cards from either beginning or end. Return maximum points you can obtain.

JavaScript Solution

function maxScore(cardPoints, k) {
    const n = cardPoints.length;
    
    // Total sum of all cards
    const totalSum = cardPoints.reduce((a, b) => a + b, 0);
    
    // If we need all cards
    if (k === n) return totalSum;
    
    // Find minimum sum subarray of size (n - k)
    const windowSize = n - k;
    let windowSum = 0;
    
    // First window
    for (let i = 0; i < windowSize; i++) {
        windowSum += cardPoints[i];
    }
    
    let minWindowSum = windowSum;
    
    // Slide window
    for (let i = windowSize; i < n; i++) {
        windowSum += cardPoints[i] - cardPoints[i - windowSize];
        minWindowSum = Math.min(minWindowSum, windowSum);
    }
    
    // Maximum points = total - minimum window
    return totalSum - minWindowSum;
}

// Usage Example:
const cardPoints = [1, 2, 3, 4, 5, 6, 1];
const k = 3;
console.log(maxScore(cardPoints, k)); // 12 (take 1+6+5 from ends)
