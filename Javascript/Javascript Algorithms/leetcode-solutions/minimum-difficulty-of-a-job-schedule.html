Minimum Difficulty of a Job Schedule

LeetCode Problem - Dynamic Programming

Problem Description

You want to schedule a list of jobs in d days. Jobs must be done in order. Each day you must do at least one job. The difficulty of a day is the maximum difficulty of a job done that day. Return the minimum difficulty of the schedule.

JavaScript Solution

function minDifficulty(jobDifficulty, d) {
    const n = jobDifficulty.length;
    
    // Not enough jobs to schedule
    if (n < d) return -1;
    
    // dp[i][day] = min difficulty to schedule jobs 0..i in day days
    const dp = new Array(n).fill(null)
        .map(() => new Array(d + 1).fill(Infinity));
    
    // Base case: first day
    let maxDiff = 0;
    for (let i = 0; i < n; i++) {
        maxDiff = Math.max(maxDiff, jobDifficulty[i]);
        dp[i][1] = maxDiff;
    }
    
    // Fill DP table
    for (let day = 2; day <= d; day++) {
        for (let i = day - 1; i < n; i++) {
            let currentMax = 0;
            // Try all possible last day partitions
            for (let j = i; j >= day - 1; j--) {
                currentMax = Math.max(currentMax, jobDifficulty[j]);
                dp[i][day] = Math.min(dp[i][day], dp[j - 1][day - 1] + currentMax);
            }
        }
    }
    
    return dp[n - 1][d];
}

// Usage Example:
const jobDifficulty = [6, 5, 4, 3, 2, 1];
const d = 2;
console.log(minDifficulty(jobDifficulty, d)); // 7 (first 5 jobs day 1, last job day 2)
