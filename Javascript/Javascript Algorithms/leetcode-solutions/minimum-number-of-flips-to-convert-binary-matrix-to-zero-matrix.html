Minimum Flips to Zero Matrix

LeetCode Problem - BFS State Space Search

Problem Description

Given a m x n binary matrix mat. In one step, you can choose one cell and flip it (0 to 1, 1 to 0) along with its 4 adjacent cells. Return minimum steps to convert matrix to all zeros.

JavaScript Solution

function minFlips(mat) {
    const m = mat.length, n = mat[0].length;
    const dirs = [[0, 0], [0, 1], [0, -1], [1, 0], [-1, 0]];
    
    // Convert matrix to bitmask
    function toBitmask(matrix) {
        let mask = 0;
        for (let i = 0; i < m; i++) {
            for (let j = 0; j < n; j++) {
                if (matrix[i][j] === 1) {
                    mask |= (1 << (i * n + j));
                }
            }
        }
        return mask;
    }
    
    // Flip cell and neighbors
    function flip(mask, x, y) {
        for (const [dx, dy] of dirs) {
            const nx = x + dx, ny = y + dy;
            if (nx >= 0 && nx < m && ny >= 0 && ny < n) {
                mask ^= (1 << (nx * n + ny));
            }
        }
        return mask;
    }
    
    const start = toBitmask(mat);
    if (start === 0) return 0;
    
    const visited = new Set();
    const queue = [[start, 0]]; // [mask, steps]
    visited.add(start);
    
    while (queue.length > 0) {
        const [mask, steps] = queue.shift();
        
        // Try flipping each cell
        for (let i = 0; i < m; i++) {
            for (let j = 0; j < n; j++) {
                const newMask = flip(mask, i, j);
                
                if (newMask === 0) return steps + 1;
                
                if (!visited.has(newMask)) {
                    visited.add(newMask);
                    queue.push([newMask, steps + 1]);
                }
            }
        }
    }
    
    return -1;
}

// Usage Example:
const mat = [[0,0],[0,1]];
console.log(minFlips(mat)); // Returns minimum flips
