Missing Element in Sorted Array

LeetCode Problem - Binary Search

Problem Description

Given a sorted array of unique integers, find the kth missing number starting from the leftmost number.

JavaScript Solution

function missingElement(nums, k) {
    // Helper: count missing numbers up to index i
    const missing = (i) => nums[i] - nums[0] - i;
    
    let left = 0, right = nums.length - 1;
    
    // If kth missing is beyond last element
    if (k > missing(right)) {
        return nums[right] + k - missing(right);
    }
    
    // Binary search for position
    while (left < right) {
        const mid = Math.floor((left + right) / 2);
        
        if (missing(mid) < k) {
            left = mid + 1;
        } else {
            right = mid;
        }
    }
    
    return nums[left - 1] + k - missing(left - 1);
}

// Usage:
console.log(missingElement([4,7,9,10], 1)); // 5
console.log(missingElement([1,2,4], 3)); // 6
