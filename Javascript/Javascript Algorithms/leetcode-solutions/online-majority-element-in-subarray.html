Online Majority Element In Subarray

LeetCode Problem - Segment Tree + Binary Search Solution

Problem Description

Design a data structure that efficiently finds the majority element of a given subarray. The majority element is an element that appears more than or equal to threshold times in the subarray.

JavaScript Solution

class MajorityChecker {
    constructor(arr) {
        this.arr = arr;
        this.pos = new Map();
        
        // Store positions of each element
        for (let i = 0; i < arr.length; i++) {
            if (!this.pos.has(arr[i])) {
                this.pos.set(arr[i], []);
            }
            this.pos.get(arr[i]).push(i);
        }
    }

    query(left, right, threshold) {
        const length = right - left + 1;
        
        // Try random sampling approach
        for (let i = 0; i < 10; i++) {
            const randomIdx = Math.floor(Math.random() * length) + left;
            const candidate = this.arr[randomIdx];
            
            if (this.getFrequency(candidate, left, right) >= threshold) {
                return candidate;
            }
        }
        
        return -1;
    }

    getFrequency(value, left, right) {
        if (!this.pos.has(value)) return 0;
        
        const positions = this.pos.get(value);
        
        // Binary search for left boundary
        const leftIdx = this.lowerBound(positions, left);
        // Binary search for right boundary
        const rightIdx = this.upperBound(positions, right);
        
        return rightIdx - leftIdx;
    }

    lowerBound(arr, target) {
        let lo = 0, hi = arr.length;
        while (lo < hi) {
            const mid = Math.floor((lo + hi) / 2);
            if (arr[mid] < target) {
                lo = mid + 1;
            } else {
                hi = mid;
            }
        }
        return lo;
    }

    upperBound(arr, target) {
        let lo = 0, hi = arr.length;
        while (lo < hi) {
            const mid = Math.floor((lo + hi) / 2);
            if (arr[mid] <= target) {
                lo = mid + 1;
            } else {
                hi = mid;
            }
        }
        return lo;
    }
}

// Usage Example:
const arr = [1, 1, 2, 2, 1];
const checker = new MajorityChecker(arr);
console.log(checker.query(0, 2, 2)); // Returns 1
