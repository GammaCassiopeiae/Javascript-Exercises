Pyramid Transition Matrix

LeetCode Problem - Backtracking DFS

Problem Description

You are stacking blocks to form a pyramid. Each block has a color represented by a letter. Given a bottom row and allowed triples, determine if you can build the pyramid to the top.

JavaScript Solution

function pyramidTransition(bottom, allowed) {
    const map = new Map();
    
    // Build adjacency map
    for (const triple of allowed) {
        const key = triple.slice(0, 2);
        if (!map.has(key)) {
            map.set(key, []);
        }
        map.get(key).push(triple[2]);
    }
    
    function canBuild(current) {
        if (current.length === 1) return true;
        
        const nextLevel = [];
        return buildNext(current, 0, '');
        
        function buildNext(row, idx, next) {
            if (idx === row.length - 1) {
                return canBuild(next);
            }
            
            const key = row.slice(idx, idx + 2);
            const candidates = map.get(key) || [];
            
            for (const c of candidates) {
                if (buildNext(row, idx + 1, next + c)) {
                    return true;
                }
            }
            
            return false;
        }
    }
    
    return canBuild(bottom);
}

// Usage Example:
const bottom = "BCD";
const allowed = ["BCC", "CDE", "CEA", "FFF"];
console.log(pyramidTransition(bottom, allowed)); // true or false
