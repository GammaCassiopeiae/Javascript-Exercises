Random Pick with Weight

LeetCode Problem - Prefix Sum + Binary Search

Problem Description

Given an array of weights, implement a function that randomly picks an index where the probability of picking index i is proportional to weights[i].

JavaScript Solution

class Solution {
    constructor(w) {
        this.prefixSum = [];
        let sum = 0;
        
        // Build prefix sum array
        for (const weight of w) {
            sum += weight;
            this.prefixSum.push(sum);
        }
        
        this.total = sum;
    }

    pickIndex() {
        // Generate random target in range [1, total]
        const target = Math.random() * this.total;
        
        // Binary search for first prefix sum > target
        let left = 0, right = this.prefixSum.length;
        
        while (left < right) {
            const mid = Math.floor((left + right) / 2);
            if (this.prefixSum[mid] <= target) {
                left = mid + 1;
            } else {
                right = mid;
            }
        }
        
        return left;
    }
}

// Usage Example:
const solution = new Solution([1, 3]);
console.log(solution.pickIndex()); // Returns 0 with 25% prob, 1 with 75% prob
