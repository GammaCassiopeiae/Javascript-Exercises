Reachable Nodes In Subdivided Graph

LeetCode Problem - Dijkstra's Algorithm

Problem Description

You are given an undirected graph with subdivided edges. Each edge can be subdivided into multiple nodes. Find the number of nodes reachable from node 0 within maxMoves steps.

JavaScript Solution

function reachableNodes(edges, maxMoves, n) {
    // Build adjacency list
    const graph = new Map();
    for (let i = 0; i < n; i++) {
        graph.set(i, []);
    }
    
    for (const [u, v, cnt] of edges) {
        graph.get(u).push([v, cnt + 1]);
        graph.get(v).push([u, cnt + 1]);
    }
    
    // Dijkstra's algorithm
    const dist = new Array(n).fill(Infinity);
    dist[0] = 0;
    
    const pq = [[0, 0]]; // [distance, node]
    
    while (pq.length > 0) {
        pq.sort((a, b) => a[0] - b[0]);
        const [d, node] = pq.shift();
        
        if (d > dist[node]) continue;
        
        for (const [neighbor, weight] of graph.get(node)) {
            const newDist = d + weight;
            if (newDist < dist[neighbor] && newDist <= maxMoves) {
                dist[neighbor] = newDist;
                pq.push([newDist, neighbor]);
            }
        }
    }
    
    // Count reachable original nodes
    let result = 0;
    for (let i = 0; i < n; i++) {
        if (dist[i] <= maxMoves) result++;
    }
    
    // Count reachable subdivided nodes
    for (const [u, v, cnt] of edges) {
        const fromU = Math.max(0, maxMoves - dist[u]);
        const fromV = Math.max(0, maxMoves - dist[v]);
        result += Math.min(cnt, fromU + fromV);
    }
    
    return result;
}

// Usage Example:
const edges = [[0,1,10],[0,2,1],[1,2,2]];
console.log(reachableNodes(edges, 6, 3)); // Returns reachable count
