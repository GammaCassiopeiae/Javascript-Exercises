Remove 9

LeetCode Problem - Base Conversion

Problem Description

Start from integer 1, remove any integer that contains 9. Find the nth positive integer in the remaining sequence.

JavaScript Solution

function newInteger(n) {
    // Convert to base-9 representation
    let result = '';
    
    while (n > 0) {
        result = (n % 9).toString() + result;
        n = Math.floor(n / 9);
    }
    
    return parseInt(result);
}

// Usage:
console.log(newInteger(9)); // 10 (sequence: 1,2,3,4,5,6,7,8,10)
console.log(newInteger(10)); // 11
