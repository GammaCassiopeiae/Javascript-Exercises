Remove K Digits

LeetCode Problem - Monotonic Stack

Problem Description

Given a non-negative integer num represented as a string, remove k digits to create the smallest possible number.

JavaScript Solution

function removeKdigits(num, k) {
    const stack = [];
    
    for (const digit of num) {
        while (k > 0 && stack.length > 0 && stack[stack.length - 1] > digit) {
            stack.pop();
            k--;
        }
        stack.push(digit);
    }
    
    // Remove remaining from end
    while (k > 0) {
        stack.pop();
        k--;
    }
    
    // Remove leading zeros
    let result = stack.join('').replace(/^0+/, '');
    
    return result || "0";
}

// Usage:
console.log(removeKdigits("1432219", 3)); // "1219"
console.log(removeKdigits("10200", 1)); // "200"
