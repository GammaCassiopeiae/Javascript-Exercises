Reverse Only Letters

LeetCode Problem - Two Pointers

Problem Description

Given a string, reverse only the English letters. All non-letter characters stay in the same position.

JavaScript Solution

function reverseOnlyLetters(s) {
    const chars = s.split('');
    let left = 0, right = chars.length - 1;
    
    while (left < right) {
        if (!/[a-zA-Z]/.test(chars[left])) {
            left++;
        } else if (!/[a-zA-Z]/.test(chars[right])) {
            right--;
        } else {
            [chars[left], chars[right]] = [chars[right], chars[left]];
            left++;
            right--;
        }
    }
    
    return chars.join('');
}

// Usage:
console.log(reverseOnlyLetters("ab-cd")); // "dc-ba"
console.log(reverseOnlyLetters("Test1ng-Leet=code-Q!")); // "Qedo1ct-eeLg=ntse-T!"
