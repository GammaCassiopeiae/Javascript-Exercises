Similar String Groups

LeetCode Problem - Union-Find / DFS

Problem Description

Two strings are similar if we can swap two letters in one to get the other. Find the number of groups where each group contains similar strings.

JavaScript Solution

function numSimilarGroups(strs) {
    const n = strs.length;
    const visited = new Set();
    let groups = 0;
    
    function isSimilar(s1, s2) {
        let diff = 0;
        for (let i = 0; i < s1.length; i++) {
            if (s1[i] !== s2[i]) diff++;
        }
        return diff <= 2;
    }
    
    function dfs(idx) {
        for (let i = 0; i < n; i++) {
            if (!visited.has(i) && isSimilar(strs[idx], strs[i])) {
                visited.add(i);
                dfs(i);
            }
        }
    }
    
    for (let i = 0; i < n; i++) {
        if (!visited.has(i)) {
            visited.add(i);
            dfs(i);
            groups++;
        }
    }
    
    return groups;
}

// Usage:
console.log(numSimilarGroups(["tars","rats","arts","star"])); // 2
