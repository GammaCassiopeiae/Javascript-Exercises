Single Number II

LeetCode Problem - Bit Manipulation

Problem Description

Given an integer array nums where every element appears three times except for one, which appears exactly once. Find the single element.

JavaScript Solution

function singleNumber(nums) {
    let ones = 0, twos = 0;
    
    for (const num of nums) {
        // Add to ones if not in twos
        ones = (ones ^ num) & ~twos;
        // Add to twos if not in ones
        twos = (twos ^ num) & ~ones;
    }
    
    return ones;
}

// Alternative: Bit counting approach
function singleNumberBitCount(nums) {
    let result = 0;
    
    for (let i = 0; i < 32; i++) {
        let sum = 0;
        for (const num of nums) {
            sum += (num >> i) & 1;
        }
        if (sum % 3 !== 0) {
            result |= (1 << i);
        }
    }
    
    return result;
}

// Usage:
console.log(singleNumber([2,2,3,2])); // 3
