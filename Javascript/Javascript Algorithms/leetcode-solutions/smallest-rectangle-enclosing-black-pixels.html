Smallest Rectangle Enclosing Black Pixels

LeetCode Problem - Binary Search on Image

Problem Description

Given an image of black ('1') and white ('0') pixels and one black pixel position, find the area of the smallest rectangle enclosing all black pixels.

JavaScript Solution

function minArea(image, x, y) {
    const m = image.length, n = image[0].length;
    
    function hasBlackInRow(row) {
        for (let j = 0; j < n; j++) {
            if (image[row][j] === '1') return true;
        }
        return false;
    }
    
    function hasBlackInCol(col) {
        for (let i = 0; i < m; i++) {
            if (image[i][col] === '1') return true;
        }
        return false;
    }
    
    // Binary search for boundaries
    function binarySearch(low, high, check, findFirst) {
        while (low < high) {
            const mid = Math.floor((low + high) / 2);
            if (check(mid) !== findFirst) {
                low = mid + 1;
            } else {
                high = mid;
            }
        }
        return low;
    }
    
    const top = binarySearch(0, x, hasBlackInRow, true);
    const bottom = binarySearch(x + 1, m, hasBlackInRow, false);
    const left = binarySearch(0, y, hasBlackInCol, true);
    const right = binarySearch(y + 1, n, hasBlackInCol, false);
    
    return (bottom - top) * (right - left);
}

// Usage:
const image = [["0","0","1","0"],["0","1","1","0"],["0","1","0","0"]];
console.log(minArea(image, 0, 2)); // 6
