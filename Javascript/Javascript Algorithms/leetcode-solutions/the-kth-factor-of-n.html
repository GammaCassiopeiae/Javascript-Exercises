The Kth Factor of N

LeetCode Problem - Math & Iteration

Problem Description

Given two positive integers n and k, return the kth factor of n when all factors are sorted in ascending order. Return -1 if n has less than k factors.

JavaScript Solution

function kthFactor(n, k) {
    let count = 0;
    
    for (let i = 1; i <= n; i++) {
        if (n % i === 0) {
            count++;
            if (count === k) return i;
        }
    }
    
    return -1;
}

// Optimized: O(sqrt(n))
function kthFactorOptimized(n, k) {
    const factors = [];
    
    for (let i = 1; i * i <= n; i++) {
        if (n % i === 0) {
            factors.push(i);
            if (i !== n / i) factors.push(n / i);
        }
    }
    
    factors.sort((a, b) => a - b);
    return factors[k - 1] || -1;
}

// Usage:
console.log(kthFactor(12, 3)); // 4 (factors: 1,2,3,4,6,12)
