Uncommon Words from Two Sentences

LeetCode Problem - Hash Map Frequency Count

Problem Description

A sentence is a string of single-space separated words. A word is uncommon if it appears exactly once in one of the sentences and does not appear in the other. Return all uncommon words.

JavaScript Solution

function uncommonFromSentences(s1, s2) {
    const freq = new Map();
    
    // Count frequency of all words
    const words1 = s1.split(' ');
    const words2 = s2.split(' ');
    
    for (const word of words1) {
        freq.set(word, (freq.get(word) || 0) + 1);
    }
    
    for (const word of words2) {
        freq.set(word, (freq.get(word) || 0) + 1);
    }
    
    // Find words that appear exactly once
    const result = [];
    for (const [word, count] of freq) {
        if (count === 1) {
            result.push(word);
        }
    }
    
    return result;
}

// Usage Example:
const s1 = "this apple is sweet";
const s2 = "this apple is sour";
console.log(uncommonFromSentences(s1, s2)); // ["sweet", "sour"]
