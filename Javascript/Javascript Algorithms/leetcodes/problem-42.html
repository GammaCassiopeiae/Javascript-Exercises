Kth Largest Element in an Array

Medium
ArraySortingHeap

Problem Description

Find the kth largest element in the array.

Example 1:

Input: nums = [3,2,1,5,6,4], k = 2
Output: 5

Constraints:

  • 1 <= k <= nums.length

Solution

QuickSelect Solution

Optimal approach for solving Kth Largest Element in an Array.

function findKthLargest(nums, k) {
  function partition(left, right, pivotIndex) {
    const pivot = nums[pivotIndex];
    [nums[pivotIndex], nums[right]] = [nums[right], nums[pivotIndex]];
    let storeIndex = left;
    for (let i = left; i < right; i++) {
      if (nums[i] > pivot) {
        [nums[storeIndex], nums[i]] = [nums[i], nums[storeIndex]];
        storeIndex++;
      }
    }
    [nums[right], nums[storeIndex]] = [nums[storeIndex], nums[right]];
    return storeIndex;
  }
  function select(left, right, kSmallest) {
    if (left === right) return nums[left];
    const pivotIndex = partition(left, right, left);
    if (kSmallest === pivotIndex) return nums[kSmallest];
    return kSmallest < pivotIndex ? select(left, pivotIndex-1, kSmallest) : select(pivotIndex+1, right, kSmallest);
  }
  return select(0, nums.length-1, k-1);
}

Complexity Analysis:

  • Time: O(n) avg
  • Space: O(1)
