javascript flood-fill.js 
function floodFill(image, sr, sc, newColor) {
  const oldColor = image[sr][sc];
  if (oldColor === newColor) return image;
  const stack = [[sr, sc]];
  while (stack.length) {
    const [r, c] = stack.pop();
    if (image[r][c] !== oldColor) continue;
    image[r][c] = newColor;
    for (const [dr, dc] of [[1,0],[-1,0],[0,1],[0,-1]]) {
      const nr = r + dr, nc = c + dc;
      if (nr >= 0 && nr < image.length && nc >= 0 && nc < image[0].length) {
        stack.push([nr, nc]);
      }
    }
  }
  return image;
}

  function dfs(node) {
    if (!node) return 0;
    const left = dfs(node.left);
    if (left === -1) return -1;
    const right = dfs(node.right);
    if (right === -1) return -1;

    if (sum > target) return;
    if (sum === target) { result.push([...curr]); return; }

function permute(nums) {
  const result = [];
  function backtrack(curr, used) {
    if (curr.length === nums.length) {
      result.push([...curr]); return;
    }
    for (let i = 0; i < nums.length; i++) {
      if (used[i]) continue;
      used[i] = true; curr.push(nums[i]);
      backtrack(curr, used);
      curr.pop(); used[i] = false;
    }
  }
  backtrack([], []);
  return result;

  for (const [name, ...emails] of accounts) {

function rightSideView(root) {
  const result = [];
  const queue = [root];
  while (queue.length) {
    const level = queue.length;
    for (let i = 0; i < level; i++) {
      const node = queue.shift();
      if (i === level - 1) result.push(node.val);
      if (node.left) queue.push(node.left);
      if (node.right) queue.push(node.right);
    }
  }
  return result;
}

javascript longest-palindromic.js 
function longestPalindrome(s) {
  if (s.length < 2) return s;
  let start = 0, max = 0;
  const expandAroundCenter = (l, r) => {
    while (l >= 0 && r < s.length && s[l] === s[r]) {
      l--; r++;
    }
    return r - l - 1;
  };
  for (let i = 0; i < s.length; i++) {
    const len1 = expandAroundCenter(i, i);
    const len2 = expandAroundCenter(i, i + 1);
    const len = Math.max(len1, len2);
    if (len > max) { max = len; start = i - Math.floor((len - 1) / 2); }
  }
  return s.substring(start, start + max);
}

function uniquePaths(m, n) {
  const dp = Array(m).fill(null).map(() => Array(n).fill(1));
  for (let i = 1; i < m; i++) {
    for (let j = 1; j < n; j++) {
      dp[i][j] = dp[i - 1][j] + dp[i][j - 1];
    }
  }
  return dp[m - 1][n - 1];
}

function buildTree(preorder, inorder) {
  const inMap = new Map(inorder.map((v, i) => [v, i]));
  function build(preSt, preEnd, inSt, inEnd) {
    if (preSt > preEnd) return null;
    const root = new TreeNode(preorder[preSt]);
    const inIdx = inMap.get(preorder[preSt]);
    const leftSize = inIdx - inSt;
    root.left = build(preSt + 1, preSt + leftSize, inSt, inIdx - 1);
    root.right = build(preSt + leftSize + 1, preEnd, inIdx + 1, inEnd);
    return root;
  }
  return build(0, preorder.length - 1, 0, inorder.length - 1);
}

function maxArea(height) {
  let maxWater = 0, left = 0, right = height.length - 1;
  while (left < right) {
    const water = (right - left) * Math.min(height[left], height[right]);
    maxWater = Math.max(maxWater, water);
    if (height[left] < height[right]) left++;
    else right--;
  }
  return maxWater;
}

function letterCombinations(digits) {
  const map = { 2: 'abc', 3: 'def', 4: 'ghi', 5: 'jkl', 6: 'mno', 7: 'pqrs', 8: 'tuv', 9: 'wxyz' };
  const result = [];
  if (!digits) return result;
  const backtrack = (index, current) => {
    if (index === digits.length) {
      result.push(current);
      return;
    }
    const letters = map[digits[index]];
    for (const letter of letters) {
      backtrack(index + 1, current + letter);
    }
  };
  backtrack(0, '');
  return result;
}

Given an m x n grid of characters and a string word, return true if the word exists in the grid (can move up, down, left, right).

function exist(board, word) {
  const dfs = (i, j, index) => {
    if (index === word.length) return true;
    if (i < 0 || i >= board.length || j < 0 || j >= board[0].length) return false;
    if (board[i][j] !== word[index]) return false;
    const temp = board[i][j];
    board[i][j] = '*';
    const found = dfs(i + 1, j, index + 1) || dfs(i - 1, j, index + 1) || dfs(i, j + 1, index + 1) || dfs(i, j - 1, index + 1);
    board[i][j] = temp;
    return found;
  };
  for (let i = 0; i < board.length; i++) {
    for (let j = 0; j < board[0].length; j++) {
      if (dfs(i, j, 0)) return true;
    }
  }
  return false;
}
