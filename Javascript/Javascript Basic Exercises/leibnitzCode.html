function kombinacija(n, k) {
    if (k === 0 || k === n) return 1;
    let produkt = 1;
    for (let i = 1; i <= k; i++) {
        produkt = produkt * (n - i + 1) / i;
    }
    return produkt;
}

function generirajTrikotnik() {
    const štVrstic = parseInt(document.getElementById('rowsInput').value);
    const kontejner = document.getElementById('triangleContainer');
    kontejner.innerHTML = '';

    for (let n = 0; n < štVrstic; n++) {
        const vrsticaDiv = document.createElement('div');
        vrsticaDiv.className = 'row';

        for (let k = 0; k <= n; k++) {
            const imenovalec = (n + 1) * kombinacija(n, k);
            const ulomek = `1/${imenovalec}`;

            const celicaSpan = document.createElement('span');
            celicaSpan.className = 'cell';
            celicaSpan.innerText = ulomek;
            
            vrsticaDiv.appendChild(celicaSpan);
        }
        kontejner.appendChild(vrsticaDiv);
    }
}
