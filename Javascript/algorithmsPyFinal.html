def catalan(n):
    # C(n) = sum(C(i) * C(n - 1 - i)).
    if n < 0:
        raise ValueError("n must be non-negative")
    dp = [0] * (n + 1)
    dp[0] = 1
    for nodes in range(1, n + 1):
        for left in range(nodes):
            dp[nodes] += dp[left] * dp[nodes - 1 - left]
    return dp[n]

def max_gold(mine):
    if not mine or not mine[0]:
        return 0
    rows, cols = len(mine), len(mine[0])
    dp = [row[:] for row in mine]
    for col in range(cols - 2, -1, -1):
        for row in range(rows):
            right = dp[row][col + 1]
            up = dp[row - 1][col + 1] if row else 0
            down = dp[row + 1][col + 1] if row + 1 < rows else 0
            dp[row][col] += max(right, up, down)
    return max(dp[row][0] for row in range(rows))

Permutation Coefficient

Compute P(n, k), the number of ordered selections of k items from n items.

def permutation_coefficient(n, k):
    if n < 0 or k < 0 or k > n:
        return 0
    dp = [0] * (k + 1)
    dp[0] = 1
    for items in range(1, n + 1):
        # P(items, chosen) = P(items-1, chosen)
        # + chosen * P(items-1, chosen-1).
        for chosen in range(min(items, k), 0, -1):
            dp[chosen] = dp[chosen] + chosen * dp[chosen - 1]
    return dp[k]

def knapsack_01(weights, values, capacity):

        # Descending capacity prevents reusing an item.

Coin Change Problem

def coin_change(coins, amount):
    # Minimum number of coins.
    minimum = [amount + 1] * (amount + 1)
    minimum[0] = 0
    for total in range(1, amount + 1):
        for coin in coins:
            if coin <= total:
                minimum[total] = min(
                    minimum[total], minimum[total - coin] + 1
                )
    fewest = minimum[amount] if minimum[amount] <= amount else -1

    # Number of combinations (coin order does not matter).
    ways = [0] * (amount + 1)
    ways[0] = 1
    for coin in coins:
        for total in range(coin, amount + 1):
            ways[total] += ways[total - coin]
    return fewest, ways[amount]

def max_box_stack_height(boxes):
    rotations = []
    for height, width, depth in boxes:
        for base_a, base_b, tall in (
            (width, depth, height), (height, depth, width), (height, width, depth)
        ):
            rotations.append((max(base_a, base_b),
                               min(base_a, base_b), tall))
    rotations.sort(key=lambda box: box[0] * box[1], reverse=True)
    heights = [box[2] for box in rotations]
    for current in range(len(rotations)):
        for below in range(current):
            if (rotations[below][0] > rotations[current][0]
                    and rotations[below][1] > rotations[current][1]):
                heights[current] = max(heights[current], heights[below] + rotations[current][2])
    return max(heights, default=0)

def lcs_length(first, second):
    memo = {}

    def solve(i, j):
        if i == len(first) or j == len(second):
            return 0
        if (i, j) not in memo:
            if first[i] == second[j]:
                memo[i, j] = 1 + solve(i + 1, j + 1)
            else:
                memo[i, j] = max(solve(i + 1, j), solve(i, j + 1))

def edit_distance(source, target):
    previous = list(range(len(target) + 1))
    for i, source_char in enumerate(source, 1):
        current = [i]
        for j, target_char in enumerate(target, 1):
            cost = 0 if source_char == target_char else 1
            current.append(min(
                current[j - 1] + 1,
                previous[j] + 1,
                previous[j - 1] + cost,
            ))
        previous = current
    return previous[-1]

            saved = dp[index]

def regex_match(text, pattern):
    memo = {}

    def solve(i, j):
        if (i, j) in memo:
            return memo[i, j]
        if j == len(pattern):
            return i == len(text)
        first_matches = i < len(text) and (pattern[j] in (".", text[i]))
        if j + 1 < len(pattern) and pattern[j + 1] == "*":
            result = solve(i, j + 2) or (first_matches and solve(i + 1, j))

            result = first_matches and solve(i + 1, j + 1)
        memo[i, j] = result
        return result

Convex Hull: Graham Scan

Find the outer boundary of a point set by sorting points and removing right turns.

def graham_scan(points):
    points = sorted(set(points))
    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])
    lower = []
    for point in points:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], point) <= 0: lower.pop()
        lower.append(point)
    upper = []
    for point in reversed(points):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], point) <= 0: upper.pop()
        upper.append(point)
    return lower[:-1] + upper[:-1]
