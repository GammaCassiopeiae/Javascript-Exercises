🫧 BUBBLE SORT
Compare adjacent elements and swap them if they are in the wrong order. Repeat until sorted.
Bubble Sort – For Loop
Classic nested for loops. The outer loop counts passes, the inner loop compares adjacent pairs and swaps.
function bubbleSortFor(arr) {
    const n = arr.length;

    for (let i = 0; i < n - 1; i++) {
        for (let j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                let temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }

    return arr;
}
Bubble Sort – While Loop
Same idea but uses a while loop to track whether any swaps happened. Stops early when already sorted.
function bubbleSortWhile(arr) {
    let swapped = true;

    while (swapped) {
        swapped = false;

        for (let i = 0; i < arr.length - 1; i++) {
            if (arr[i] > arr[i + 1]) {
                let temp = arr[i];
                arr[i] = arr[i + 1];
                arr[i + 1] = temp;
                swapped = true;
            }
        }
    }

    return arr;
}
Bubble Sort – Recursive
Each recursive call does one full pass (pushing the largest element to the end), then recurses on the smaller portion.
function bubbleSortRecursive(arr, n) {
    if (n === undefined) n = arr.length;

    if (n === 1) return arr;

    for (let i = 0; i < n - 1; i++) {
        if (arr[i] > arr[i + 1]) {
            let temp = arr[i];
            arr[i] = arr[i + 1];
            arr[i + 1] = temp;
        }
    }

    return bubbleSortRecursive(arr, n - 1);
}
GNOME SORT
Like a garden gnome sorting flower pots — walks forward comparing, walks back when out of order.
Gnome Sort – For Loop
Uses a for loop to walk through the array. When a swap happens, the index moves back to check previous elements.
function gnomeSortFor(arr) {
    const n = arr.length;

    for (let i = 0; i < n; ) {
        if (i === 0 || arr[i - 1] <= arr[i]) {
            i++;
        } else {
            let temp = arr[i];
            arr[i] = arr[i - 1];
            arr[i - 1] = temp;
            i--;
        }
    }

    return arr;
}
Gnome Sort – While Loop
A while loop with an explicit index. The gnome walks forward or backward one step at a time.
function gnomeSortWhile(arr) {
    let i = 0;

    while (i < arr.length) {
        if (i === 0 || arr[i - 1] <= arr[i]) {
            i++;
        } else {
            let temp = arr[i];
            arr[i] = arr[i - 1];
            arr[i - 1] = temp;
            i--;
        }
    }

    return arr;
}
Gnome Sort – Recursive
Each recursive call either moves forward or swaps and steps back. Simple tail-recursive logic.
function gnomeSortRec(arr, i) {
    if (i === undefined) i = 0;

    if (i === arr.length) return arr;

    if (i === 0 || arr[i - 1] <= arr[i]) {
        return gnomeSortRec(arr, i + 1);
    } else {
        let temp = arr[i];
        arr[i] = arr[i - 1];
        arr[i - 1] = temp;
        return gnomeSortRec(arr, i - 1);
    }
}
BOGO SORT
The worst sorting algorithm ever — randomly shuffle until the array happens to be sorted. For learning only!
Bogo Sort – For Loop
Outer for loop counts shuffle attempts. Each iteration shuffles and checks if sorted. Brace yourself — it may take a while.
function bogoSortFor(arr) {
    const maxAttempts = 100000;

    for (let attempt = 0; attempt < maxAttempts; attempt++) {
        let isSorted = true;

        for (let i = 0; i < arr.length - 1; i++) {
            if (arr[i] > arr[i + 1]) {
                isSorted = false;
                break;
            }
        }

        if (isSorted) return arr;

        for (let i = arr.length - 1; i > 0; i--) {
            let j = Math.floor(Math.random() * (i + 1));
            let temp = arr[i];
            arr[i] = arr[j];
            arr[j] = temp;
        }
    }

    return arr;
}
Bogo Sort – While Loop
A while loop keeps shuffling until the array is sorted. Simple, clean, and hilariously inefficient.
function bogoSortWhile(arr) {
    let attempts = 0;
    let maxAttempts = 100000;

    while (attempts < maxAttempts) {
        let isSorted = true;

        for (let i = 0; i < arr.length - 1; i++) {
            if (arr[i] > arr[i + 1]) {
                isSorted = false;
                break;
            }
        }

        if (isSorted) return arr;

        for (let i = arr.length - 1; i > 0; i--) {
            let j = Math.floor(Math.random() * (i + 1));
            let temp = arr[i];
            arr[i] = arr[j];
            arr[j] = temp;
        }

        attempts++;
    }

    return arr;
}
Bogo Sort – Recursive
Each recursive call checks if sorted, shuffles if not, then calls itself again. The recursion tree can be deep — use small arrays!
function bogoSortRec(arr, attempts) {
    if (attempts === undefined) attempts = 0;

    let maxAttempts = 100000;

    if (attempts >= maxAttempts) return arr;

    let isSorted = true;

    for (let i = 0; i < arr.length - 1; i++) {
        if (arr[i] > arr[i + 1]) {
            isSorted = false;
            break;
        }
    }

    if (isSorted) return arr;

    for (let i = arr.length - 1; i > 0; i--) {
        let j = Math.floor(Math.random() * (i + 1));
        let temp = arr[i];
        arr[i] = arr[j];
        arr[j] = temp;
    }

    return bogoSortRec(arr, attempts + 1);
}
INSERTION SORT
Pick each element and insert it into its correct position among the already-sorted elements.
Insertion Sort – For Loop
Outer for loop picks the next element. Inner for loop shifts larger elements to the right to make room.
function insertionSortFor(arr) {
    for (let i = 1; i < arr.length; i++) {
        let key = arr[i];
        let j = i - 1;

        for (; j >= 0; j--) {
            if (arr[j] > key) {
                arr[j + 1] = arr[j];
            } else {
                break;
            }
        }

        arr[j + 1] = key;
    }

    return arr;
}
Insertion Sort – While Loop
A while loop shifts elements right until the correct spot is found for the current element.
function insertionSortWhile(arr) {
    for (let i = 1; i < arr.length; i++) {
        let key = arr[i];
        let j = i - 1;

        while (j >= 0 && arr[j] > key) {
            arr[j + 1] = arr[j];
            j--;
        }

        arr[j + 1] = key;
    }

    return arr;
}
Insertion Sort – Recursive
Recursively sorts the first n-1 elements, then inserts the last element into the correct position.
function insertionSortRec(arr, n) {
    if (n === undefined) n = arr.length;

    if (n <= 1) return arr;

    insertionSortRec(arr, n - 1);

    let key = arr[n - 1];
    let j = n - 2;

    while (j >= 0 && arr[j] > key) {
        arr[j + 1] = arr[j];
        j--;
    }

    arr[j + 1] = key;

    return arr;
}
SELECTION SORT
Find the smallest element in the unsorted part and swap it into position. Repeat until done.
Selection Sort – For Loop
Outer for loop marks the boundary. Inner for loop finds the minimum in the unsorted part and swaps it forward.
function selectionSortFor(arr) {
    for (let i = 0; i < arr.length - 1; i++) {
        let minIdx = i;

        for (let j = i + 1; j < arr.length; j++) {
            if (arr[j] < arr[minIdx]) {
                minIdx = j;
            }
        }

        if (minIdx !== i) {
            let temp = arr[i];
            arr[i] = arr[minIdx];
            arr[minIdx] = temp;
        }
    }

    return arr;
}
Selection Sort – While Loop
Uses a while loop to scan for the minimum. Same logic, different loop structure.
function selectionSortWhile(arr) {
    let i = 0;

    while (i < arr.length - 1) {
        let minIdx = i;
        let j = i + 1;

        while (j < arr.length) {
            if (arr[j] < arr[minIdx]) {
                minIdx = j;
            }
            j++;
        }

        if (minIdx !== i) {
            let temp = arr[i];
            arr[i] = arr[minIdx];
            arr[minIdx] = temp;
        }

        i++;
    }

    return arr;
}
Selection Sort – Recursive
Finds the minimum in the unsorted part, swaps it to the front, then recurses on the rest.
function selectionSortRec(arr, start) {
    if (start === undefined) start = 0;

    if (start >= arr.length - 1) return arr;

    let minIdx = start;

    for (let j = start + 1; j < arr.length; j++) {
        if (arr[j] < arr[minIdx]) {
            minIdx = j;
        }
    }

    if (minIdx !== start) {
        let temp = arr[start];
        arr[start] = arr[minIdx];
        arr[minIdx] = temp;
    }

    return selectionSortRec(arr, start + 1);
}
BUCKET SORT
Distribute elements into buckets, sort each bucket individually, then merge them back together.
Bucket Sort – For Loop
For loop distributes elements into buckets. Each bucket is sorted with insertion sort, then merged.
function bucketSortFor(arr) {
    if (arr.length === 0) return arr;

    let min = Math.min(...arr);
    let max = Math.max(...arr);
    let bucketCount = Math.floor((max - min) / arr.length) + 1;
    let buckets = [];

    for (let i = 0; i < bucketCount; i++) {
        buckets.push([]);
    }

    for (let i = 0; i < arr.length; i++) {
        let idx = Math.floor((arr[i] - min) / arr.length);
        buckets[idx].push(arr[i]);
    }

    let result = [];

    for (let i = 0; i < buckets.length; i++) {
        buckets[i].sort(function(a, b) { return a - b; });

        for (let j = 0; j < buckets[i].length; j++) {
            result.push(buckets[i][j]);
        }
    }

    return result;
}
Bucket Sort – While Loop
While loop iterates through elements to distribute, then while loops merge the sorted buckets.
function bucketSortWhile(arr) {
    if (arr.length === 0) return arr;

    let min = Math.min(...arr);
    let max = Math.max(...arr);
    let bucketCount = Math.floor((max - min) / arr.length) + 1;
    let buckets = [];

    for (let i = 0; i < bucketCount; i++) {
        buckets.push([]);
    }

    let i = 0;

    while (i < arr.length) {
        let idx = Math.floor((arr[i] - min) / arr.length);
        buckets[idx].push(arr[i]);
        i++;
    }

    let result = [];
    let b = 0;

    while (b < buckets.length) {
        buckets[b].sort(function(a, c) { return a - c; });

        let j = 0;

        while (j < buckets[b].length) {
            result.push(buckets[b][j]);
            j++;
        }

        b++;
    }

    return result;
}
Bucket Sort – Recursive
Recursively sorts each bucket after distributing elements. Base case is a bucket with 0 or 1 elements.
function bucketSortRec(arr) {
    if (arr.length <= 1) return arr;

    let min = Math.min(...arr);
    let max = Math.max(...arr);
    let bucketCount = Math.floor((max - min) / arr.length) + 1;
    let buckets = [];

    for (let i = 0; i < bucketCount; i++) {
        buckets.push([]);
    }

    for (let i = 0; i < arr.length; i++) {
        let idx = Math.floor((arr[i] - min) / arr.length);
        buckets[idx].push(arr[i]);
    }

    let result = [];

    for (let i = 0; i < buckets.length; i++) {
        let sorted = bucketSortRec(buckets[i]);

        for (let j = 0; j < sorted.length; j++) {
            result.push(sorted[j]);
        }
    }

    return result;
}
MATRIX SORT
Flatten a 2D matrix into a 1D array, sort it, then reshape it back into the original dimensions.
Matrix Sort – For Loop
For loops flatten the matrix, sort the flat array with insertion sort, then for loops reshape it back.
function matrixSortFor(matrix) {
    let flat = [];

    for (let i = 0; i < matrix.length; i++) {
        for (let j = 0; j < matrix[i].length; j++) {
            flat.push(matrix[i][j]);
        }
    }

    for (let i = 1; i < flat.length; i++) {
        let key = flat[i];
        let j = i - 1;

        for (; j >= 0; j--) {
            if (flat[j] > key) {
                flat[j + 1] = flat[j];
            } else {
                break;
            }
        }

        flat[j + 1] = key;
    }

    let rows = matrix.length;
    let cols = matrix[0].length;
    let idx = 0;
    let result = [];

    for (let i = 0; i < rows; i++) {
        result.push([]);

        for (let j = 0; j < cols; j++) {
            result[i].push(flat[idx]);
            idx++;
        }
    }

    return result;
}
Matrix Sort – While Loop
While loops flatten, sort with insertion sort, and reshape. Same logic, different loop style.
function matrixSortWhile(matrix) {
    let flat = [];
    let i = 0;

    while (i < matrix.length) {
        let j = 0;

        while (j < matrix[i].length) {
            flat.push(matrix[i][j]);
            j++;
        }

        i++;
    }

    for (let k = 1; k < flat.length; k++) {
        let key = flat[k];
        let m = k - 1;

        while (m >= 0 && flat[m] > key) {
            flat[m + 1] = flat[m];
            m--;
        }

        flat[m + 1] = key;
    }

    let rows = matrix.length;
    let cols = matrix[0].length;
    let idx = 0;
    let result = [];
    let r = 0;

    while (r < rows) {
        result.push([]);
        let c = 0;

        while (c < cols) {
            result[r].push(flat[idx]);
            idx++;
            c++;
        }

        r++;
    }

    return result;
}
Matrix Sort – Recursive
Recursively flattens, then recursively sorts the flat array, then reshapes back into a matrix.
function flattenRec(matrix, i, j, flat) {
    if (i === undefined) i = 0;
    if (j === undefined) j = 0;
    if (flat === undefined) flat = [];

    if (i >= matrix.length) return flat;

    if (j >= matrix[i].length) {
        return flattenRec(matrix, i + 1, 0, flat);
    }

    flat.push(matrix[i][j]);

    return flattenRec(matrix, i, j + 1, flat);
}

function insertionSortRec(arr, n) {
    if (n === undefined) n = arr.length;
    if (n <= 1) return arr;

    insertionSortRec(arr, n - 1);

    let key = arr[n - 1];
    let j = n - 2;

    while (j >= 0 && arr[j] > key) {
        arr[j + 1] = arr[j];
        j--;
    }

    arr[j + 1] = key;

    return arr;
}

function matrixSortRec(matrix) {
    let flat = flattenRec(matrix);

    insertionSortRec(flat);

    let rows = matrix.length;
    let cols = matrix[0].length;
    let result = [];
    let idx = 0;

    for (let i = 0; i < rows; i++) {
        result.push([]);

        for (let j = 0; j < cols; j++) {
            result[i].push(flat[idx]);
            idx++;
        }
    }

    return result;
}
MERGE SORT
Divide the array in half, recursively sort each half, then merge the sorted halves together.
Merge Sort – For Loop
Uses for loops to split the array at the midpoint and merge two sorted halves back together.
function mergeSortFor(arr) {
    if (arr.length <= 1) return arr;

    let mid = Math.floor(arr.length / 2);
    let left = arr.slice(0, mid);
    let right = arr.slice(mid);

    left = mergeSortFor(left);
    right = mergeSortFor(right);

    let result = [];
    let i = 0;
    let j = 0;

    for (; i < left.length && j < right.length; ) {
        if (left[i] <= right[j]) {
            result.push(left[i]);
            i++;
        } else {
            result.push(right[j]);
            j++;
        }
    }

    for (; i < left.length; i++) {
        result.push(left[i]);
    }

    for (; j < right.length; j++) {
        result.push(right[j]);
    }

    return result;
}
Merge Sort – While Loop
While loops handle the merging step. The recursion splits, the while loops compare and merge.
function mergeSortWhile(arr) {
    if (arr.length <= 1) return arr;

    let mid = Math.floor(arr.length / 2);
    let left = arr.slice(0, mid);
    let right = arr.slice(mid);

    left = mergeSortWhile(left);
    right = mergeSortWhile(right);

    let result = [];
    let i = 0;
    let j = 0;

    while (i < left.length && j < right.length) {
        if (left[i] <= right[j]) {
            result.push(left[i]);
            i++;
        } else {
            result.push(right[j]);
            j++;
        }
    }

    while (i < left.length) {
        result.push(left[i]);
        i++;
    }

    while (j < right.length) {
        result.push(right[j]);
        j++;
    }

    return result;
}
Merge Sort – Recursive
The classic recursive approach: split in half, sort each half recursively, merge the results.
function mergeSortRec(arr) {
    if (arr.length <= 1) return arr;

    let mid = Math.floor(arr.length / 2);
    let left = arr.slice(0, mid);
    let right = arr.slice(mid);

    left = mergeSortRec(left);
    right = mergeSortRec(right);

    return merge(left, right);
}

function merge(left, right) {
    let result = [];
    let i = 0;
    let j = 0;

    while (i < left.length && j < right.length) {
        if (left[i] <= right[j]) {
            result.push(left[i]);
            i++;
        } else {
            result.push(right[j]);
            j++;
        }
    }

    while (i < left.length) {
        result.push(left[i]);
        i++;
    }

    while (j < right.length) {
        result.push(right[j]);
        j++;
    }

    return result;
}
QUICKSORT
Pick a pivot, partition elements into smaller and larger groups, then sort each group.
Quicksort – For Loop
For loop partitions elements around a pivot. Everything smaller goes left, everything larger goes right.
function quicksortFor(arr) {
    if (arr.length <= 1) return arr;

    let pivot = arr[arr.length - 1];
    let left = [];
    let right = [];

    for (let i = 0; i < arr.length - 1; i++) {
        if (arr[i] < pivot) {
            left.push(arr[i]);
        } else {
            right.push(arr[i]);
        }
    }

    let sortedLeft = quicksortFor(left);
    let sortedRight = quicksortFor(right);

    let result = [];

    for (let i = 0; i < sortedLeft.length; i++) {
        result.push(sortedLeft[i]);
    }

    result.push(pivot);

    for (let i = 0; i < sortedRight.length; i++) {
        result.push(sortedRight[i]);
    }

    return result;
}
Quicksort – While Loop
While loop builds the left and right partitions. Same partition logic, different loop style.
function quicksortWhile(arr) {
    if (arr.length <= 1) return arr;

    let pivot = arr[arr.length - 1];
    let left = [];
    let right = [];
    let i = 0;

    while (i < arr.length - 1) {
        if (arr[i] < pivot) {
            left.push(arr[i]);
        } else {
            right.push(arr[i]);
        }
        i++;
    }

    let sortedLeft = quicksortWhile(left);
    let sortedRight = quicksortWhile(right);

    let result = [];
    let k = 0;

    while (k < sortedLeft.length) {
        result.push(sortedLeft[k]);
        k++;
    }

    result.push(pivot);
    k = 0;

    while (k < sortedRight.length) {
        result.push(sortedRight[k]);
        k++;
    }

    return result;
}
Quicksort – Recursive
The classic recursive quicksort: pick pivot, partition, recurse on both sides. Clean and simple.
function quicksortRec(arr) {
    if (arr.length <= 1) return arr;

    let pivot = arr[arr.length - 1];
    let left = [];
    let right = [];

    for (let i = 0; i < arr.length - 1; i++) {
        if (arr[i] < pivot) {
            left.push(arr[i]);
        } else {
            right.push(arr[i]);
        }
    }

    return [
        ...quicksortRec(left),
        pivot,
        ...quicksortRec(right)
    ];
}
MIXED SORT (Cocktail Shaker Sort)
Bubble sort that goes back and forth — left to right, then right to left. Pushes big items right AND small items left.
Mixed Sort – For Loop
Alternating for loops sweep left-to-right and right-to-left. Like bubble sort but bidirectional.
function mixedSortFor(arr) {
    let start = 0;
    let end = arr.length - 1;

    for (let pass = 0; pass < arr.length; pass++) {
        let swapped = false;

        for (let i = start; i < end; i++) {
            if (arr[i] > arr[i + 1]) {
                let temp = arr[i];
                arr[i] = arr[i + 1];
                arr[i + 1] = temp;
                swapped = true;
            }
        }

        end--;

        for (let i = end; i > start; i--) {
            if (arr[i - 1] > arr[i]) {
                let temp = arr[i];
                arr[i] = arr[i - 1];
                arr[i - 1] = temp;
                swapped = true;
            }
        }

        start++;

        if (!swapped) break;
    }

    return arr;
}
Mixed Sort – While Loop
While loop controls the bidirectional passes. Swaps until no more swaps happen in either direction.
function mixedSortWhile(arr) {
    let start = 0;
    let end = arr.length - 1;
    let swapped = true;

    while (swapped) {
        swapped = false;

        for (let i = start; i < end; i++) {
            if (arr[i] > arr[i + 1]) {
                let temp = arr[i];
                arr[i] = arr[i + 1];
                arr[i + 1] = temp;
                swapped = true;
            }
        }

        end--;

        for (let i = end; i > start; i--) {
            if (arr[i - 1] > arr[i]) {
                let temp = arr[i];
                arr[i] = arr[i - 1];
                arr[i - 1] = temp;
                swapped = true;
            }
        }

        start++;
    }

    return arr;
}
Mixed Sort – Recursive
Recursive version does one bidirectional pass, then recurses on the remaining unsorted portion.
function mixedSortRec(arr, start, end) {
    if (start === undefined) start = 0;
    if (end === undefined) end = arr.length - 1;

    if (start >= end) return arr;

    let swapped = false;

    for (let i = start; i < end; i++) {
        if (arr[i] > arr[i + 1]) {
            let temp = arr[i];
            arr[i] = arr[i + 1];
            arr[i + 1] = temp;
            swapped = true;
        }
    }

    for (let i = end; i > start; i--) {
        if (arr[i - 1] > arr[i]) {
            let temp = arr[i];
            arr[i] = arr[i - 1];
            arr[i - 1] = temp;
            swapped = true;
        }
    }

    if (!swapped) return arr;

    return mixedSortRec(arr, start + 1, end - 1);
}
STAR & CHARACTER PATTERNS
Practice loops and grid systems — these build the logic foundation for everything else
Square Pattern
Print an N × N grid of stars. The most basic pattern — two nested loops.
function squareFor(n) {
    let result = "";

    for (let i = 0; i < n; i++) {
        for (let j = 0; j < n; j++) {
            result += "* ";
        }
        result += "\n";
    }

    return result;
}
Square Pattern – While
Same square using while loops for rows and columns.
function squareWhile(n) {
    let result = "";
    let i = 0;

    while (i < n) {
        let j = 0;

        while (j < n) {
            result += "* ";
            j++;
        }

        result += "\n";
        i++;
    }

    return result;
}

class MinHeap {
    constructor() {
        this.heap = [];
    }

    insert(val) {
        this.heap.push(val);
        this.bubbleUp(this.heap.length - 1);
    }

    bubbleUp(idx) {
        while (idx > 0) {
            let parent = Math.floor((idx - 1) / 2);

            if (this.heap[idx] < this.heap[parent]) {
                [this.heap[idx], this.heap[parent]] =
                    [this.heap[parent], this.heap[idx]];
                idx = parent;
            } else break;
        }
    }

    extractMin() {
        let min = this.heap[0];
        this.heap[0] = this.heap[this.heap.length - 1];
        this.heap.pop();
        this.sinkDown(0);

        return min;
    }

    sinkDown(idx) {
        let length = this.heap.length;

        while (true) {
            let left = 2 * idx + 1;
            let right = 2 * idx + 2;
            let smallest = idx;

            if (left < length &&
                this.heap[left] < this.heap[smallest]) {
                smallest = left;
            }

            if (right < length &&
                this.heap[right] < this.heap[smallest]) {
                smallest = right;
            }

            if (smallest !== idx) {
                [this.heap[idx], this.heap[smallest]] =
                    [this.heap[smallest], this.heap[idx]];
                idx = smallest;

        }

function topKFrequentFor(arr, k) {
    let freq = {};

    for (let i = 0; i < arr.length; i++) {
        freq[arr[i]] = (freq[arr[i]] || 0) + 1;
    }

    let entries = Object.entries(freq);

    entries.sort((a, b) => b[1] - a[1]);

    let result = [];

    for (let i = 0; i < k && i < entries.length; i++) {
        result.push(Number(entries[i][0]));
    }

    return result;
}

class TreeNode {

function inorderFor(root) {
    if (root === null) return [];

    let result = [];

    while (current !== null || stack.length > 0) {

            stack.push(current);
            current = current.left;
        }

        current = stack.pop();
        result.push(current.value);
        current = current.right;
    }

    return result;
}

Delete in BST
Remove elements while keeping the tree sorted.
function deleteBST(root, value) {
    if (root === null) return null;

    if (value < root.value) {
        root.left = deleteBST(root.left, value);
    } else if (value > root.value) {
        root.right = deleteBST(root.right, value);
    } else {
        if (root.left === null) return root.right;
        if (root.right === null) return root.left;

        let minNode = root.right;

        while (minNode.left !== null) {
            minNode = minNode.left;
        }

        root.value = minNode.value;
        root.right = deleteBST(root.right, minNode.value);
    }

    return root;
}
Delete BST – While
While loop — find node and handle 0/1/2 children cases.
function deleteBSTWhile(root, value) {
    if (root === null) return null;

    let current = root;
    let parent = null;

    while (current !== null && current.value !== value) {
        parent = current;

        if (value < current.value) {
            current = current.left;
        } else {
            current = current.right;
        }
    }

    if (current === null) return root;

    if (current.left === null && current.right === null) {
        if (parent === null) return null;

        if (parent.left === current) parent.left = null;
        else parent.right = null;
    } else if (current.left === null) {
        if (parent === null) return current.right;

        if (parent.left === current) parent.left = current.right;
        else parent.right = current.right;
    } else if (current.right === null) {
        if (parent === null) return current.left;

        if (parent.left === current) parent.left = current.left;
        else parent.right = current.left;
    } else {
        let successor = current.right;
        let successorParent = current;

        while (successor.left !== null) {
            successorParent = successor;
            successor = successor.left;
        }

        current.value = successor.value;

        if (successorParent === current) {
            successorParent.right = successor.right;
        } else {
            successorParent.left = successor.right;
        }
    }

    return root;
}
Check if Valid BST
Verify that a tree meets all Binary Search Tree rules.
function isValidBST(root) {
    function validate(node, min, max) {
        if (node === null) return true;

        if (node.value <= min || node.value >= max) {
            return false;
        }

        return validate(node.left, min, node.value) &&
               validate(node.right, node.value, max);
    }

    return validate(root, -Infinity, Infinity);
}

function isValidBSTWhile(root) {
    if (root === null) return true;
    let stack = [];
    let prev = -Infinity;
    let current = root;

    while (current !== null || stack.length > 0) {
        while (current !== null) {
            stack.push(current);
            current = current.left;
        }

        current = stack.pop();

        if (current.value <= prev) return false;

        prev = current.value;
        current = current.right;
    }

    return true;
}

Graph Representation
Build graphs using Adjacency Matrices and Adjacency Lists.
class Graph {
    constructor(vertices) {
        this.vertices = vertices;
        this.adjMatrix = [];

        for (let i = 0; i < vertices; i++) {
            this.adjMatrix.push(new Array(vertices).fill(0));
        }

        this.adjList = {};

        for (let i = 0; i < vertices; i++) {
            this.adjList[i] = [];
        }
    }

    addEdge(u, v) {
        this.adjMatrix[u][v] = 1;
        this.adjMatrix[v][u] = 1;

        this.adjList[u].push(v);
        this.adjList[v].push(u);
    }

    toString() {
        let matrix = this.adjMatrix
            .map(row => row.join(" ")).join("\n");

        let list = Object.entries(this.adjList)
            .map(([k, v]) => k + " → " + v.join(", "))
            .join("\n");

        return "Adjacency Matrix:\n" + matrix +
               "\n\nAdjacency List:\n" + list;
    }
}

Breadth-First Search (BFS)
Explore a graph layer by layer from a starting point.
function bfsFor(adjList, start) {
    let visited = new Set();
    let queue = [start];
    let result = [];

    visited.add(start);

    while (queue.length > 0) {
        let node = queue.shift();
        result.push(node);

        for (let i = 0; i < adjList[node].length; i++) {
            let neighbor = adjList[node][i];

            if (!visited.has(neighbor)) {
                visited.add(neighbor);
                queue.push(neighbor);
            }
        }
    }

    return result;
}

Depth-First Search (DFS)
Explore a graph by going as deep as possible down a path first.
function dfsFor(adjList, start) {
    let visited = new Set();
    let stack = [start];
    let result = [];

    while (stack.length > 0) {
        let node = stack.pop();

        if (!visited.has(node)) {
            visited.add(node);
            result.push(node);

            for (let i = adjList[node].length - 1; i >= 0; i--) {
                if (!visited.has(adjList[node][i])) {
                    stack.push(adjList[node][i]);
                }
            }
        }
    }

    return result;
}

Cycle Detection (Undirected)
Check if an undirected graph contains a closed loop path.
function hasCycleFor(adjList, vertices) {
    let visited = new Set();

    for (let i = 0; i < vertices; i++) {
        if (!visited.has(i)) {
            let stack = [[i, -1]];
            visited.add(i);

            while (stack.length > 0) {
                let [node, parent] = stack.pop();

                for (let j = 0; j < adjList[node].length; j++) {
                    let neighbor = adjList[node][j];

                    if (!visited.has(neighbor)) {
                        visited.add(neighbor);
                        stack.push([neighbor, node]);
                    } else if (neighbor !== parent) {
                        return true;
                    }
                }
            }
        }
    }

    return false;
}

Dijkstra's Algorithm
Find the shortest path from a starting point to all other locations.
function dijkstraFor(graph, start) {
    let dist = {};

    for (let node in graph) {
        dist[node] = Infinity;
    }

    dist[start] = 0;
    let visited = new Set();
    let queue = [[0, start]];

    while (queue.length > 0) {
        queue.sort((a, b) => a[0] - b[0]);
        let [d, u] = queue.shift();

        if (visited.has(u)) continue;

        visited.add(u);

        for (let v in graph[u]) {
            if (!visited.has(v)) {
                let newDist = d + graph[u][v];

                if (newDist < dist[v]) {
                    dist[v] = newDist;
                    queue.push([newDist, v]);
                }
            }
        }
    }

    return dist;
}

    let i = 0;

        dist[nodes[i]] = Infinity;
        i++;
    }

        queue.sort((a, b) => a[0] - b[0]);

Topological Sort
Order tasks linearly based on dependencies (Prerequisites problem).
function topoSortFor(vertices, edges) {
    let adjList = {};
    let inDegree = {};

    for (let i = 0; i < vertices; i++) {
        adjList[i] = [];
        inDegree[i] = 0;
    }

    for (let i = 0; i < edges.length; i++) {
        adjList[edges[i][0]].push(edges[i][1]);
        inDegree[edges[i][1]]++;
    }

    let queue = [];

    for (let node in inDegree) {
        if (inDegree[node] === 0) {
            queue.push(Number(node));
        }
    }

    let result = [];

    while (queue.length > 0) {
        let node = queue.shift();
        result.push(node);

        for (let i = 0; i < adjList[node].length; i++) {
            let neighbor = adjList[node][i];
            inDegree[neighbor]--;

            if (inDegree[neighbor] === 0) {
                queue.push(neighbor);
            }
        }
    }

    return result.length === vertices ? result : "Cycle detected";
}

Number of Islands
Count isolated clusters of connected grid items.
function numIslandsFor(grid) {
    if (grid.length === 0) return 0;
    let rows = grid.length;
    let cols = grid[0].length;
    let count = 0;

    for (let r = 0; r < rows; r++) {
        for (let c = 0; c < cols; c++) {
            if (grid[r][c] === 1) {
                count++;
                let stack = [[r, c]];

                while (stack.length > 0) {
                    let [cr, cc] = stack.pop();

                    if (cr >= 0 && cr < rows &&
                        cc >= 0 && cc < cols &&
                        grid[cr][cc] === 1) {
                        grid[cr][cc] = 0;
                        stack.push([cr - 1, cc]);
                        stack.push([cr + 1, cc]);
                        stack.push([cr, cc - 1]);
                        stack.push([cr, cc + 1]);
                    }
                }
            }
        }
    }

    return count;
}

class UnionFind {
    constructor(n) {
        this.parent = [];
        this.rank = [];

        for (let i = 0; i < n; i++) {
            this.parent[i] = i;
            this.rank[i] = 0;
        }
    }

    find(x) {
        while (this.parent[x] !== x) {
            this.parent[x] = this.parent[this.parent[x]];
            x = this.parent[x];
        }
        return x;
    }

    union(x, y) {
        let rootX = this.find(x);
        let rootY = this.find(y);

        if (rootX === rootY) return false;

        if (this.rank[rootX] < this.rank[rootY]) {
            this.parent[rootX] = rootY;
        } else if (this.rank[rootX] > this.rank[rootY]) {
            this.parent[rootY] = rootX;
        } else {
            this.parent[rootY] = rootX;
            this.rank[rootX]++;
        }

        return true;
    }

    connected(x, y) {
        return this.find(x) === this.find(y);
    }
}

        let rootY = this.find(y);

        if (rootX === rootY) return false;

        if (this.rank[rootX] < this.rank[rootY]) {
            this.parent[rootX] = rootY;
        } else if (this.rank[rootX] > this.rank[rootY]) {
            this.parent[rootY] = rootX;
        } else {
            this.parent[rootY] = rootX;
            this.rank[rootX]++;
        }

        return true;
    }

    connected(x, y) {
        return this.find(x) === this.find(y);
    }
