51. Segment Tree
Handles range queries and point updates efficiently in O(log N).
class SegTree {
    constructor(arr) {
        this.n = arr.length;
        this.tree = Array(4 * this.n).fill(0);
        this.build(arr, 1, 0, this.n - 1);
    }
    build(arr, node, s, e) {
        if (s === e) { this.tree[node] = arr[s]; return; }
        let m = (s + e) >> 1;
        this.build(arr, node*2, s, m);
        this.build(arr, node*2+1, m+1, e);
        this.tree[node] = this.tree[node*2] + this.tree[node*2+1];
    }
    query(l, r, node, s, e) {
        if (r < s || l > e) return 0;
        if (l <= s && e <= r) return this.tree[node];
        let m = (s + e) >> 1;
        return this.query(l, r, node*2, s, m) +
               this.query(l, r, node*2+1, m+1, e);
    }

56. BFS & DFS
Foundation for graph traversals and connectivity checks.
function bfs(adj, start) {
    const visited = new Set([start]);
    const queue = [start], order = [];
    while (queue.length) {
        const u = queue.shift();
        order.push(u);
        for (const v of adj[u])
            if (!visited.has(v)) { visited.add(v); queue.push(v); }
    }
    return order;
}
function dfs(adj, start, visited = new Set()) {
    visited.add(start);
    const order = [start];
    for (const v of adj[start])
        if (!visited.has(v)) order.push(...dfs(adj, v, visited));
    return order;
}
