01 Matrix

Distance from each cell to the nearest zero

Difficulty Medium
Time O(m · n)
Space O(m · n)
Pattern Multi-Source BFS

Given an m × n binary matrix, return the distance — number of 4-direction steps — from each cell to the nearest 0.

Seed a BFS queue with every zero cell and expand outward, writing the step count as you go.

JAVASCRIPT SOLUTION
function updateMatrix(mat) {
    const rows = mat.length;
    const cols = mat[0].length;
    const queue = [];
    const INF = rows * cols;

    // Seed zeros, mark ones as "infinity"
    for (let r = 0; r < rows; r++) {
        for (let c = 0; c < cols; c++) {
            if (mat[r][c] === 0) {
                queue.push([r, c]);
            } else {
                mat[r][c] = INF;
            }
        }
    }

    const dirs = [[1, 0], [-1, 0], [0, 1], [0, -1]];

    // Multi-source BFS
    for (let i = 0; i < queue.length; i++) {
        const [r, c] = queue[i];
        for (const [dr, dc] of dirs) {
            const nr = r + dr;
            const nc = c + dc;
            if (nr < 0 || nr >= rows || nc < 0 || nc >= cols) continue;

            if (mat[nr][nc] > mat[r][c] + 1) {
                mat[nr][nc] = mat[r][c] + 1;
                queue.push([nr, nc]);
            }
        }
    }

    return mat;
}

// Example
console.log(updateMatrix([[0, 0, 0], [0, 1, 0], [1, 1, 1]]));
// → [[0,0,0],[0,1,0],[1,2,1]]
