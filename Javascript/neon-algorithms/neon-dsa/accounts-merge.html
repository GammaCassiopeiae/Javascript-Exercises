Accounts Merge

Merge accounts sharing a common email

Difficulty Medium
Time O(A · log A)
Space O(A · E)
Pattern Union-Find / DFS

Each account is [name, email1, ...]. Merge accounts that share at least one email, returning each merged account's name followed by its sorted unique emails.

Union accounts by shared emails (or DFS over email adjacency), then collect each component's email set.

JAVASCRIPT SOLUTION
// Union-Find over account indices
class DSU {
    constructor(n) {
        this.parent = Array.from({ length: n }, (_, i) => i);
    }
    find(x) {
        if (this.parent[x] !== x) {
            this.parent[x] = this.find(this.parent[x]);
        }
        return this.parent[x];
    }
    union(a, b) {
        this.parent[this.find(a)] = this.find(b);
    }
}

function accountsMerge(accounts) {
    const dsu = new DSU(accounts.length);
    const emailToIndex = new Map();

    // Union accounts that share an email
    for (let i = 0; i < accounts.length; i++) {
        for (let j = 1; j < accounts[i].length; j++) {
            const email = accounts[i][j];
            if (emailToIndex.has(email)) {
                dsu.union(i, emailToIndex.get(email));
            } else {
                emailToIndex.set(email, i);
            }
        }
    }

    // Group emails by root account
    const groups = new Map();
    for (const [email, index] of emailToIndex) {
        const root = dsu.find(index);
        if (!groups.has(root)) {
            groups.set(root, new Set());
        }
        groups.get(root).add(email);
    }

    const result = [];
    for (const [index, emails] of groups) {
        const sorted = [...emails].sort();
        result.push([accounts[index][0], ...sorted]);
    }

    return result;
}

// Example
const accounts = [
    ["John", "j1@mail.com", "j2@mail.com"],
    ["John", "j2@mail.com", "j3@mail.com"],
    ["Mary", "m@mail.com"]
];
console.log(accountsMerge(accounts));
// → [["John", "j1@mail.com", "j2@mail.com", "j3@mail.com"], ["Mary", "m@mail.com"]]

> accountsMerge(accounts) → merged accounts with unique sorted emails
