Basic Calculator

Evaluate a string with +, -, and parentheses

Difficulty Hard
Time O(n)
Space O(n)
Pattern Stack

Evaluate an expression string containing digits, +, -, parentheses, and spaces (unary minus allowed).

Track a running sign; when a parenthesis opens, push the accumulated result and sign, reset, and restore on ).

JAVASCRIPT SOLUTION
function calculate(s) {
    const stack = [];
    let result = 0;
    let sign = 1;
    let num = 0;

    for (let i = 0; i < s.length; i++) {
        const char = s[i];

        if (char >= '0' && char <= '9') {
            num = num * 10 + (char.charCodeAt(0) - 48);
        } else if (char === '+') {
            result += sign * num;
            num = 0;
            sign = 1;
        } else if (char === '-') {
            result += sign * num;
            num = 0;
            sign = -1;
        } else if (char === '(') {
            stack.push(result);
            stack.push(sign);
            result = 0;
            sign = 1;
        } else if (char === ')') {
            result += sign * num;
            num = 0;
            result *= stack.pop();   // restore sign
            result += stack.pop();   // restore result
        }
    }

    return result + sign * num;
}

// Example
console.log(calculate("1 + 1"));             // 2
console.log(calculate(" 2-1 + 2 "));          // 3
console.log(calculate("(1+(4+5+2)-3)+(6+8)")); // 23
