Construct Binary Tree from Preorder and Inorder

Rebuild a unique tree from two traversals

Difficulty Medium
Time O(n)
Space O(n)
Pattern DFS · Recursion

Given the preorder and inorder traversals of a unique binary tree, reconstruct it.

Preorder's first value is the root; its position in inorder separates the left and right subtrees; recurse with index ranges.

JAVASCRIPT SOLUTION
function buildTree(preorder, inorder) {
    const inIndex = new Map();
    for (let i = 0; i < inorder.length; i++) {
        inIndex.set(inorder[i], i);
    }

    let preIdx = 0;

    function build(inLeft, inRight) {
        if (inLeft > inRight) {
            return null;
        }

        const val = preorder[preIdx++];
        const node = new TreeNode(val);
        const split = inIndex.get(val);

        node.left = build(inLeft, split - 1);
        node.right = build(split + 1, inRight);

        return node;
    }

    return build(0, inorder.length - 1);
}

// TreeNode class assumed:
// function TreeNode(val) { this.val = val; }

// Example
// preorder  = [3, 9, 20, 15, 7]
// inorder   = [9, 3, 15, 20, 7]
// →
//       3
//      / \
//     9  20
//       /  \
//      15   7

> buildTree([3,9,20,15,7], [9,3,15,20,7]) → root of rebuilt tree
