Decode Ways

Count ways to decode a digit string into letters

Difficulty Medium
Time O(n)
Space O(1)
Pattern Dynamic Programming

A message maps 'A'→1 … 'Z'→26. Given a digit string, count the number of ways to decode it.

At each position, add the ways from decoding one digit (1–9) and, when valid, two digits (10–26).

JAVASCRIPT SOLUTION
function numDecodings(s) {
    if (s.length === 0 || s[0] === '0') {
        return 0;
    }

    let prev = 1;  // ways before previous char
    let curr = 1;  // ways up to current char

    for (let i = 1; i < s.length; i++) {
        let next = 0;
        const one = Number(s[i]);
        const two = Number(s.slice(i - 1, i + 1));

        if (one >= 1 && one <= 9) {
            next += curr;
        }
        if (two >= 10 && two <= 26) {
            next += prev;
        }

        prev = curr;
        curr = next;
    }

    return curr;
}

// Example
console.log(numDecodings("12"));   // 2  ("AB" or "L")
console.log(numDecodings("226"));  // 3  ("BZ", "VF", "BBF")
console.log(numDecodings("06"));   // 0
