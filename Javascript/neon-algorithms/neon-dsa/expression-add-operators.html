Expression Add Operators

Insert +, -, * to reach the target

Difficulty Hard
Time O(4ⁿ)
Space O(n)
Pattern Backtracking

Insert +, -, or * between digits of num (no leading zeros on operands) so the expression equals target. Return all strings.

Backtrack while tracking the running total and the last operand; multiplication must undo the previous add before applying the product.

JAVASCRIPT SOLUTION
function addOperators(num, target) {
    const result = [];

    function backtrack(index, path, value, prev) {
        if (index === num.length) {
            if (value === target) result.push(path);
            return;
        }

        for (let i = index; i < num.length; i++) {
            const substr = num.slice(index, i + 1);
            if (substr.length > 1 && substr[0] === '0') break;  // no leading zero

            const current = Number(substr);

            if (index === 0) {
                backtrack(i + 1, substr, current, current);
            } else {
                backtrack(i + 1, `${path}+${substr}`, value + current, current);
                backtrack(i + 1, `${path}-${substr}`, value - current, -current);
                // undo the previous operand, then multiply
                backtrack(i + 1, `${path}*${substr}`,
                           value - prev + prev * current, prev * current);
            }
        }
    }

    backtrack(0, "", 0, 0);
    return result;
}

// Example
console.log(addOperators("123", 6));
// → ["1*2*3", "1+2+3"]
