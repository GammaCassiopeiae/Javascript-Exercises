Find All Anagrams in a String

Locate every start index of an anagram window

Difficulty Medium
Time O(n)
Space O(1)
Pattern Sliding Window

Given strings s and p, return all start indices where p's anagram appears as a substring of s.

Slide a window of length p across s while maintaining a character-frequency difference that reaches zero at each match.

JAVASCRIPT SOLUTION
function findAnagrams(s, p) {
    const result = [];
    if (s.length < p.length) return result;

    const pCount = new Map();
    const sCount = new Map();

    for (const char of p) {
        pCount.set(char, (pCount.get(char) || 0) + 1);
    }

    let matches = 0;

    for (let i = 0; i < s.length; i++) {
        // add character entering the window
        sCount.set(s[i], (sCount.get(s[i]) || 0) + 1);
        if (sCount.get(s[i]) === pCount.get(s[i])) matches++;

        // remove character leaving the window
        const left = i - p.length;
        if (left >= 0) {
            const outChar = s[left];
            if (sCount.get(outChar) === pCount.get(outChar)) matches--;
            sCount.set(outChar, sCount.get(outChar) - 1);
        }

        if (matches === pCount.size) {
            result.push(i - p.length + 1);
        }
    }

    return result;
}

// Example
console.log(findAnagrams("cbaebabacd", "abc"));
// → [0, 6]

console.log(findAnagrams("abab", "ab"));
// → [0, 1, 2]
