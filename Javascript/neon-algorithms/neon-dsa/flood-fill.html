Flood Fill

Change the color of connected same-colored cells

Difficulty Easy
Time O(n)
Space O(n)
Pattern DFS / BFS

Given an m × n grid and a starting pixel, replace the color of all connected pixels that share the starting pixel's original color.

Depth-first search visits 4-directional neighbors while the original color matches.

JAVASCRIPT SOLUTION
function floodFill(image, sr, sc, color) {
    const startColor = image[sr][sc];
    if (startColor === color) {
        return image;
    }

    const rows = image.length;
    const cols = image[0].length;

    function dfs(r, c) {
        if (r < 0 || r >= rows || c < 0 || c >= cols) {
            return;
        }
        if (image[r][c] !== startColor) {
            return;
        }

        image[r][c] = color;

        dfs(r + 1, c);
        dfs(r - 1, c);
        dfs(r, c + 1);
        dfs(r, c - 1);
    }

    dfs(sr, sc);
    return image;
}

// Example
const image = [
    [1, 1, 1],
    [1, 1, 0],
    [1, 0, 1]
];
console.log(floodFill(image, 1, 1, 2));
// → [[2,2,2],[2,2,0],[2,0,1]]
