Longest Palindromic Substring

Find the longest palindrome inside the string

Difficulty Medium
Time O(n²)
Space O(1)
Pattern Expand Around Center

Given a string s, return the longest palindromic substring in s.

Treat every index (and gap between indexes) as a palindrome center and expand outward while the characters match.

JAVASCRIPT SOLUTION
function longestPalindrome(s) {
    if (s.length < 1) return "";

    let start = 0;
    let maxLen = 0;

    function expandAroundCenter(left, right) {
        while (left >= 0 && right < s.length &&
               s[left] === s[right]) {
            left--;
            right++;
        }
        return right - left - 1;  // length of palindrome
    }

    for (let i = 0; i < s.length; i++) {
        const oddLen = expandAroundCenter(i, i);        // single center
        const evenLen = expandAroundCenter(i, i + 1);   // double center
        const len = Math.max(oddLen, evenLen);

        if (len > maxLen) {
            maxLen = len;
            start = i - Math.floor((len - 1) / 2);
        }
    }

    return s.slice(start, start + maxLen);
}

// Example
console.log(longestPalindrome("babad")); // "bab" or "aba"
console.log(longestPalindrome("cbbd"));  // "bb"

> longestPalindrome("babad") → "bab"

> longestPalindrome("cbbd") → "bb"
