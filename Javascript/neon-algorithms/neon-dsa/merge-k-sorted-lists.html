Merge k Sorted Lists

Merge k sorted linked lists into one

Difficulty Hard
Time O(n log k)
Space O(k)
Pattern Merge Sort / Heap

Given an array of k sorted linked lists, merge them into one sorted list.

Repeatedly pairwise-merge using the standard two-list merge, halving the number of lists each round — divide and conquer.

JAVASCRIPT SOLUTION
function mergeTwoLists(list1, list2) {
    const dummy = new ListNode();
    let tail = dummy;

    while (list1 !== null && list2 !== null) {
        if (list1.val < list2.val) {
            tail.next = list1;
            list1 = list1.next;
        } else {
            tail.next = list2;
            list2 = list2.next;
        }
        tail = tail.next;
    }

    tail.next = list1 === null ? list2 : list1;
    return dummy.next;
}

function mergeKLists(lists) {
    if (lists.length === 0) {
        return null;
    }

    while (lists.length > 1) {
        const next = [];
        for (let i = 0; i < lists.length; i += 2) {
            const a = lists[i];
            const b = i + 1 < lists.length ? lists[i + 1] : null;
            next.push(mergeTwoLists(a, b));
        }
        lists = next;
    }

    return lists[0];
}

// ListNode class assumed:
// function ListNode(val, next) { ... }

// Example: merge [[1,4,5],[1,3,4],[2,6]]
// → 1 → 1 → 2 → 3 → 4 → 4 → 5 → 6
