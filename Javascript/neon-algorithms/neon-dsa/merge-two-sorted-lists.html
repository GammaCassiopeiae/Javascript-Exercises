Merge Two Sorted Lists

Merge two sorted linked lists into one

Difficulty Easy
Time O(n+m)
Space O(n+m)
Pattern Linked List

Merge two sorted linked lists list1 and list2 into one sorted list by splicing nodes together.

Use a dummy head sentinel and advance the smaller head at each step.

JAVASCRIPT SOLUTION
function ListNode(val, next) {
    this.val = (val === undefined ? 0 : val);
    this.next = (next === undefined ? null : next);
}

function mergeTwoLists(list1, list2) {
    const dummy = new ListNode();
    let tail = dummy;

    while (list1 !== null && list2 !== null) {
        if (list1.val < list2.val) {
            tail.next = list1;
            list1 = list1.next;
        } else {
            tail.next = list2;
            list2 = list2.next;
        }
        tail = tail.next;
    }

    // Attach remaining nodes
    tail.next = list1 === null ? list2 : list1;

    return dummy.next;
}

// Example: list1 = 1→3→5, list2 = 2→4→6
// Result: 1→2→3→4→5→6

> mergeTwoLists([1,3,5], [2,4,6]) → 1 → 2 → 3 → 4 → 5 → 6
