Minimum Window Substring

Smallest window containing all characters of t

Difficulty Hard
Time O(n)
Space O(m)
Pattern Sliding Window

Given strings s and t, return the smallest substring of s that contains every character of t (with counts) or "".

Expand the right edge until all requirements are met, then shrink the left edge while keeping the window valid, tracking the shortest.

JAVASCRIPT SOLUTION
function minWindow(s, t) {
    const need = new Map();
    for (const char of t) {
        need.set(char, (need.get(char) || 0) + 1);
    }

    let required = need.size;      // distinct chars still unmet
    let left = 0;
    let minLen = Infinity;
    let start = 0;

    for (let right = 0; right < s.length; right++) {
        const char = s[right];

        if (need.has(char)) {
            need.set(char, need.get(char) - 1);
            if (need.get(char) === 0) required--;
        }

        while (required === 0) {
            if (right - left + 1 < minLen) {
                minLen = right - left + 1;
                start = left;
            }

            const outChar = s[left];
            if (need.has(outChar)) {
                need.set(outChar, need.get(outChar) + 1);
                if (need.get(outChar) > 0) required++;
            }
            left++;
        }
    }

    return minLen === Infinity ? "" : s.slice(start, start + minLen);
}

// Example
console.log(minWindow("ADOBECODEBANC", "ABC")); // "BANC"
console.log(minWindow("a", "a"));               // "a"
console.log(minWindow("a", "aa"));              // ""

> minWindow("ADOBECODEBANC", "ABC") → "BANC"

> minWindow("a", "aa") → ""
