Quick Sort

Divide and conquer with pivot partitioning

Category Sorting
Time O(n log n) avg
Space O(log n)
Pattern Partitioning

Sort an array by choosing a pivot, partitioning so smaller elements sit left and larger right, then recursing on both halves in place.

Lomuto partitioning (below) places the pivot at its final index; average O(n log n), worst case O(n²) with poor pivots.

JAVASCRIPT SOLUTION
function quickSort(arr, left = 0, right = arr.length - 1) {
    if (left >= right) {
        return arr;
    }

    const pivotIndex = partition(arr, left, right);

    quickSort(arr, left, pivotIndex - 1);
    quickSort(arr, pivotIndex + 1, right);

    return arr;
}

function partition(arr, left, right) {
    const pivot = arr[right];
    let store = left;

    for (let i = left; i < right; i++) {
        if (arr[i] < pivot) {
            [arr[i], arr[store]] = [arr[store], arr[i]];
            store++;
        }
    }

    [arr[store], arr[right]] = [arr[right], arr[store]];
    return store;
}

// Example
console.log(quickSort([3, 6, 8, 10, 1, 2, 1]));
// → [1, 1, 2, 3, 6, 8, 10]

> quickSort([3,6,8,10,1,2,1]) → [1,1,2,3,6,8,10]
