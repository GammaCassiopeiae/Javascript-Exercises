Remove Invalid Parentheses

All strings with the fewest removals to balance

Difficulty Hard
Time O(2ⁿ)
Space O(n)
Pattern BFS / Backtracking

Given a string with parentheses and letters, remove the fewest parentheses to make it valid, returning every such string.

BFS deletes one character per level, tracking visited states; the first level that yields a valid expression is the answer set.

JAVASCRIPT SOLUTION
function isValid(str) {
    let balance = 0;
    for (const char of str) {
        if (char === '(') balance++;
        if (char === ')') balance--;
        if (balance < 0) return false;
    }
    return balance === 0;
}

function removeInvalidParentheses(s) {
    const result = [];
    const visited = new Set([s]);
    const queue = [s];

    while (queue.length) {
        const size = queue.length;
        const foundThisLevel = [];

        for (let i = 0; i < size; i++) {
            const current = queue.shift();

            if (isValid(current)) {
                foundThisLevel.push(current);
            } else if (foundThisLevel.length === 0) {
                // remove one character, enqueue candidates
                for (let j = 0; j < current.length; j++) {
                    if (current[j] !== '(' && current[j] !== ')') continue;
                    const next = current.slice(0, j) + current.slice(j + 1);
                    if (!visited.has(next)) {
                        visited.add(next);
                        queue.push(next);
                    }
                }
            }
        }

        if (foundThisLevel.length > 0) {
            return foundThisLevel;
        }
    }

    return [];
}

// Example
console.log(removeInvalidParentheses("()())()"));
// → ["(())()", "()()()"]

> removeInvalidParentheses("()())()") → ["(())()", "()()()"]
