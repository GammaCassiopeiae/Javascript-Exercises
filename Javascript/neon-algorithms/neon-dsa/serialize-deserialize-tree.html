Serialize and Deserialize Binary Tree

Encode a tree to a string and back

Difficulty Hard
Time O(n)
Space O(n)
Pattern DFS · Preorder

Design algorithms to serialize a binary tree into a string and deserialize that string back into the identical tree.

Preorder traversal writes null for missing children; deserialization replays the same preorder, using an index pointer to rebuild nodes.

JAVASCRIPT SOLUTION
function serialize(root) {
    const parts = [];

    function dfs(node) {
        if (node === null) {
            parts.push("null");
            return;
        }
        parts.push(String(node.val));
        dfs(node.left);
        dfs(node.right);
    }

    dfs(root);
    return parts.join(",");
}

function deserialize(data) {
    const values = data.split(",");
    let index = 0;

    function dfs() {
        const val = values[index++];
        if (val === "null") {
            return null;
        }

        const node = new TreeNode(Number(val));
        node.left = dfs();
        node.right = dfs();
        return node;
    }

    return dfs();
}

// TreeNode class assumed:
// function TreeNode(val) { this.val = val; this.left = null; }

// Example
const tree = new TreeNode(1);
tree.left = new TreeNode(2);
tree.right = new TreeNode(3);
tree.right.left = new TreeNode(4);

const s = serialize(tree);   // "1,2,null,null,3,4,null,null,null"
const copy = deserialize(s); // identical tree

> serialize(tree) → "1,2,null,null,3,4,null,null,null"

> deserialize(...) → identical tree
