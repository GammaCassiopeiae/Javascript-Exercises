String to Integer atoi

Parse a string into a 32-bit integer

Difficulty Medium
Time O(n)
Space O(1)
Pattern String Parsing

Implement atoi: ignore leading whitespace, read an optional sign, then digits. Clamp to the 32-bit signed range [-2³¹, 2³¹−1] and ignore trailing non-digits.

One forward scan with a sign flag; stop as soon as a non-digit appears.

JAVASCRIPT SOLUTION
function myAtoi(s) {
    let i = 0;
    let sign = 1;

    // 1. skip leading whitespace
    while (s[i] === ' ') i++;

    // 2. optional sign
    if (s[i] === '+' || s[i] === '-') {
        sign = s[i] === '-' ? -1 : 1;
        i++;
    }

    // 3. build the number
    let value = 0;
    const MIN = -(2 ** 31);
    const MAX = 2 ** 31 - 1;

    while (i < s.length && s[i] >= '0' && s[i] <= '9') {
        value = value * 10 + (s[i].charCodeAt(0) - 48);
        i++;

        if (sign * value > MAX) return MAX;
        if (sign * value < MIN) return MIN;
    }

    return sign * value;
}

// Example
console.log(myAtoi("   -42"));      // -42
console.log(myAtoi("4193 with words")); // 4193
console.log(myAtoi("words and 987"));   // 0
console.log(myAtoi("-91283472332"));    // -2147483648 (clamped)
