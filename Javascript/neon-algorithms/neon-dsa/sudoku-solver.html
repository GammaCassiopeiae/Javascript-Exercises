Sudoku Solver

Fill the board with backtracking

Difficulty Hard
Time O(9^empty)
Space O(81)
Pattern Backtracking

Fill a 9×9 Sudoku board in place, keeping digits 1–9 unique per row, column, and each 3×3 box.

Find an empty cell, try each valid digit, recurse, and undo on failure. Lazy bitmask validation keeps each check O(1).

JAVASCRIPT SOLUTION
function solveSudoku(board) {
    function isValid(r, c, num) {
        for (let i = 0; i < 9; i++) {
            if (board[r][i] === num) return false;   // row
            if (board[i][c] === num) return false;   // column
        }

        const boxRow = Math.floor(r / 3) * 3;
        const boxCol = Math.floor(c / 3) * 3;
        for (let i = boxRow; i < boxRow + 3; i++) {
            for (let j = boxCol; j < boxCol + 3; j++) {
                if (board[i][j] === num) return false;   // 3×3 box
            }
        }

        return true;
    }

    function backtrack() {
        for (let r = 0; r < 9; r++) {
            for (let c = 0; c < 9; c++) {
                if (board[r][c] !== '.') continue;  // empty cell?

                for (let num = 1; num <= 9; num++) {
                    const digit = String(num);
                    if (isValid(r, c, digit)) {
                        board[r][c] = digit;
                        if (backtrack()) {
                            return true;
                        }
                        board[r][c] = '.';
                    }
                }
                return false;  // no digit fits
            }
        }
        return true;  // all cells filled
    }

    backtrack();
    return board;
}
