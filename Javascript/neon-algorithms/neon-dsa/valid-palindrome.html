Valid Palindrome

Check if a phrase reads the same forward and backward

Difficulty Easy
Time O(n)
Space O(1)
Pattern Two Pointers

A phrase is a palindrome if it reads the same forward and backward after converting to lowercase and removing all non-alphanumeric characters.

Move two pointers inward while skipping non-alphanumeric characters.

JAVASCRIPT SOLUTION
function isPalindrome(s) {
    let left = 0;
    let right = s.length - 1;

    while (left < right) {
        // Skip non-alphanumeric characters
        while (left < right && !isAlphanumeric(s[left])) left++;
        while (left < right && !isAlphanumeric(s[right])) right--;

        if (s[left].toLowerCase() !== s[right].toLowerCase()) {
            return false;
        }

        left++;
        right--;
    }

    return true;
}

function isAlphanumeric(char) {
    return /^[a-z0-9]$/i.test(char);
}

// Example
console.log(isPalindrome("A man, a plan, a canal: Panama")); // true
console.log(isPalindrome("race a car"));                     // false

> isPalindrome("A man, a plan, a canal: Panama") → true

> isPalindrome("race a car") → false
