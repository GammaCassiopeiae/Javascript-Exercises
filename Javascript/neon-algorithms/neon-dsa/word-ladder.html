Word Ladder

Shortest transformation from beginWord to endWord

Difficulty Hard
Time O(n · L²)
Space O(n)
Pattern BFS

Transform beginWord into endWord, changing one letter at a time, where every intermediate word must exist in wordList. Return the shortest sequence length (or 0).

BFS over words — each step, try every single-letter mutation of the current word and expand through unvisited dictionary words.

JAVASCRIPT SOLUTION
function ladderLength(beginWord, endWord, wordList) {
    const words = new Set(wordList);
    if (!words.has(endWord)) {
        return 0;
    }

    const queue = [beginWord];
    let level = 1;

    while (queue.length) {
        const size = queue.length;

        for (let i = 0; i < size; i++) {
            const current = queue.shift();

            if (current === endWord) {
                return level;
            }

            // try every single-letter change
            for (let j = 0; j < current.length; j++) {
                for (let ch = 97; ch <= 122; ch++) {  // a..z
                    const next = current.slice(0, j) +
                                 String.fromCharCode(ch) +
                                 current.slice(j + 1);

                    if (words.has(next)) {
                        queue.push(next);
                        words.delete(next);  // visited
                    }
                }
            }
        }

        level++;
    }

    return 0;
}

// Example
console.log(ladderLength("hit", "cog",
    ["hot", "dot", "dog", "lot", "log", "cog"])); // 5
console.log(ladderLength("hit", "cog",
    ["hot", "dot", "dog", "lot", "log"]));         // 0

> ladderLength("hit", "cog", [...]) → 5 ("hit → hot → dot → dog → cog")

> ladderLength("hit", "cog", [...]) → 0 (no path)
